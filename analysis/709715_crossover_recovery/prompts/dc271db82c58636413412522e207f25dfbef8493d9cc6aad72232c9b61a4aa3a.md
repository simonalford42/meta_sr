"""
Synthesizes the age-regularized replacement strategy (favor removing the
oldest genome) with a quality-aware tie-break so that, among genomes of
similar age, the search preferentially discards the *worse* performer
rather than blindly picking whichever happens to be oldest.

Motivation: pure age-regularized survival (both parents) guarantees healthy
turnover and protects young, still-unevaluated offspring, but it is blind to
fitness — a mediocre-but-old member can be kept alive far too long relative
to peers, or a decent old member can be evicted purely due to birth order
even while a worse, slightly younger member persists. This is visible in
the trace above: complexity keeps climbing with vanishing loss gains,
suggesting bloated members are not being pruned efficiently.

Approach:
1. Rank all eligible (non-excluded) members by age (older -> higher rank)
   and by cost (worse cost -> higher rank).
2. Combine the two ranks into a single score using a tunable weight that
   keeps age as the dominant signal (preserving the original
   age-regularized character and its diversity/turnover guarantees) while
   letting cost break ties and nudge borderline decisions towards removing
   under-performing members.
3. Return the index with the highest combined score, i.e., the oldest and
   (secondarily) worst member, excluding any indices in `exclude_indices`.
"""
function age_and_cost_regularized_survival_gen19_0(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    n = pop.n
    eligible = [i for i in 1:n if !(i in exclude_indices)]
    @assert !isempty(eligible) "No eligible members to replace"

    # Extract raw birth and cost values for eligible members
    births = [pop.members[i].birth for i in eligible]
    costs = [pop.members[i].cost for i in eligible]

    # Rank by age: oldest (smallest birth) gets highest age-rank
    age_order = sortperm(births)  # ascending birth -> first is oldest
    age_rank = similar(age_order, Float64)
    for (r, idx) in enumerate(age_order)
        age_rank[idx] = (length(eligible) - r + 1) / length(eligible)  # oldest -> 1.0
    end

    # Rank by cost: worst (largest cost) gets highest cost-rank
    cost_order = sortperm(costs)  # ascending cost -> last is worst
    cost_rank = similar(cost_order, Float64)
    for (r, idx) in enumerate(cost_order)
        cost_rank[idx] = r / length(eligible)  # worst -> 1.0
    end

    # Combine ranks: age dominates (weight 0.75) but cost breaks ties and
    # nudges the decision toward removing poor performers among similarly-aged peers.
    age_weight = 0.75
    combined_score = age_weight .* age_rank .+ (1 - age_weight) .* cost_rank

    # Pick eligible member with highest combined score (oldest & worst-biased)
    best_local = argmax(combined_score)
    return eligible[best_local]
end

"""
    adaptive_pareto_niche_survival_gen25_0(pop::Population{T,L,N}, options::AbstractOptions; exclude_indices::Vector{Int}=Int[])::Int where {T,L,N}

Custom survival operator that balances age-regularization with Pareto frontier protection
and complexity-niche desaturation to select the optimal population member for replacement.

# Rationale & Heuristics:
1. **Immediate Purge of Non-Finite Individuals**:
   Members with `NaN` or `Inf` loss/cost provide zero evolutionary value and are prioritized
   for immediate replacement, freeing computational capacity.
2. **Pareto Frontier Protection**:
   Standard Age-Regularized Evolution (ARE) indiscriminately eliminates the oldest individuals,
   frequently destroying critical non-dominated Pareto stepping stones (best loss for a given
   complexity). This operator assigns a significant survival bonus (culling discount) to
   non-dominated individuals.
3. **Complexity-Niche Desaturation**:
   When multiple individuals share the same complexity bin, structural diversity decreases.
   A logarithmic congestion penalty is applied to overcrowded complexity bins to encourage
   exploration across diverse expression sizes.
4. **Age-Regularized Decay**:
   Older individuals receive higher base replacement priority, preserving the continuous
   population renewal and anti-bloat dynamics of regularized evolution.
"""
function adaptive_pareto_niche_survival_gen25_0(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T,L,N}
    n = pop.n
    if n == 0
        return 1
    end

    # Step 1: Pre-filter valid candidates not present in exclude_indices
    # Identify min and max birth timestamps across all members to normalize age
    min_birth = typemax(pop.members[1].birth)
    max_birth = typemin(pop.members[1].birth)

    @inbounds for i in 1:n
        b = pop.members[i].birth
        if b < min_birth
            min_birth = b
        end
        if b > max_birth
            max_birth = b
        end
    end

    birth_range = Float64(max_birth - min_birth + 1)

    # Step 2: Compute complexities, check finiteness, and compute complexity histogram
    complexities = Vector{Int}(undef, n)
    is_finite_member = Vector{Bool}(undef, n)
    max_c = 0

    @inbounds for i in 1:n
        c = compute_complexity(pop.members[i], options)
        complexities[i] = c
        if c > max_c
            max_c = c
        end
        loss_val = pop.members[i].loss
        cost_val = pop.members[i].cost
        is_finite_member[i] = isfinite(loss_val) && isfinite(cost_val)
    end

    # Count population density in each complexity niche
    niche_counts = zeros(Int, max_c)
    @inbounds for i in 1:n
        c = complexities[i]
        if c >= 1 && c <= max_c
            niche_counts[c] += 1
        end
    end

    # Step 3: Identify Pareto-dominated members
    # An individual is dominated if another finite individual has (<= complexity AND < loss)
    # or (< complexity AND <= loss).
    is_dominated = falses(n)
    @inbounds for i in 1:n
        if !is_finite_member[i]
            is_dominated[i] = true
            continue
        end
        c_i = complexities[i]
        loss_i = pop.members[i].loss

        for j in 1:n
            if i == j || !is_finite_member[j]
                continue
            end
            c_j = complexities[j]
            loss_j = pop.members[j].loss

            if (c_j <= c_i && loss_j < loss_i) || (c_j < c_i && loss_j <= loss_i)
                is_dominated[i] = true
                break
            end
        end
    end

    # Step 4: Compute replacement score for each candidate
    # Higher replacement score = higher priority to be replaced (culled)
    best_candidate_idx = -1
    best_candidate_score = -Inf

    @inbounds for i in 1:n
        # Skip indices marked for exclusion (e.g., during crossover pairs)
        if i in exclude_indices
            continue
        end

        # Non-finite candidates are culled with highest priority, broken by age
        if !is_finite_member[i]
            # Prioritize oldest non-finite individuals
            score = 1e9 + Float64(max_birth - pop.members[i].birth + 1)
        else
            # Age factor: linearly scales from ~0.0 (newest) to 1.0 (oldest)
            age_factor = Float64(max_birth - pop.members[i].birth + 1) / birth_range

            # Niche congestion factor: increases if complexity bin is crowded
            c_i = complexities[i]
            niche_count = (c_i >= 1 && c_i <= max_c) ? niche_counts[c_i] : 1
            congestion_factor = 1.0 + log(Float64(niche_count))

            # Pareto factor: non-dominated members receive a protective discount (0.2x),
            # while dominated members receive full/accelerated replacement pressure (1.5x)
            pareto_factor = is_dominated[i] ? 1.5 : 0.2

            score = age_factor * congestion_factor * pareto_factor
        end

        if score > best_candidate_score
            best_candidate_score = score
            best_candidate_idx = i
        end
    end

    # Fallback to first available index if no candidate met criteria
    if best_candidate_idx == -1
        for i in 1:n
            if !(i in exclude_indices)
                return i
            end
        end
        return 1
    end

    return best_candidate_idx
end
