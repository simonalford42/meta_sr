"""
    dynamic_range_pseudohuber_loss_gen1_0(tree, dataset, options)

Compute a scale-aware pseudo-Huber loss that remains quadratic near an exact fit
but only linear for large residuals. The target mean and standard deviation are
estimated with Welford's algorithm, then each residual is normalized by a blend
of the global target scale and its distance from the target mean. This modestly
downweights extreme target regimes while preserving sensitivity around typical
samples. A numerically stable pseudo-Huber formula provides robustness to noise,
outliers, and explosive early-search predictions without creating a flat,
bounded objective. The loss is zero only for an exact fit.
"""
function dynamic_range_pseudohuber_loss_gen1_0(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction) || dataset.n == 0
        return L(Inf)
    end

    n = dataset.n

    # Estimate target location and scale in one pass without allocating.
    mean_y = zero(L)
    m2_y = zero(L)
    @inbounds for i in 1:n
        yi = L(dataset.y[i])
        if !isfinite(yi)
            return L(Inf)
        end

        delta = yi - mean_y
        mean_y += delta / L(i)
        m2_y += delta * (yi - mean_y)

        if !isfinite(mean_y) || !isfinite(m2_y)
            return L(Inf)
        end
    end

    target_std = sqrt(max(zero(L), m2_y / L(n)))

    # Keep constant-target datasets well-defined without imposing a unit scale.
    scale_floor = L(8) * eps(one(L)) * max(one(L), abs(mean_y))
    global_scale = max(target_std, scale_floor)

    mean_loss = zero(L)
    one_l = one(L)
    two_l = L(2)
    local_weight = L(0.25)

    @inbounds for i in 1:n
        yi = L(dataset.y[i])
        yhat = L(prediction[i])
        if !isfinite(yhat)
            return L(Inf)
        end

        # The local term prevents high-magnitude target regimes from dominating.
        denominator = global_scale + local_weight * abs(yi - mean_y)
        z = abs((yhat - yi) / denominator)
        if !isfinite(z)
            return L(Inf)
        end

        # Stable form of 2 * (sqrt(1 + z^2) - 1):
        # quadratic near zero and linear for large normalized residuals.
        h = hypot(one_l, z)
        rho = two_l * z * (z / (h + one_l))
        if !isfinite(rho)
            return L(Inf)
        end

        # Online averaging avoids a potentially overflowing raw sum.
        mean_loss += (rho - mean_loss) / L(i)
    end

    return L(mean_loss)
end

## Parent loss operator 2

# Custom Loss: mse_loss
# =====================
# Default loss: per-sample mean squared error. Behavior matches the built-in
# `_eval_loss` path on unweighted, unitless datasets, exposed as a named
# operator so the meta-evolution loop has a concrete parent to refine.
#
# For weighted datasets or unit-bearing data, the in-module `default_loss`
# (used when no operator is loaded) handles `dataset.weights` and
# `dimensional_regularization` via `_eval_loss`. This baseline assumes
# unweighted, unitless data, which matches every SRBench dataset on the
# PySR pipeline.

function mse_loss(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end
    diff = prediction .- dataset.y
    return L(sum(abs2, diff) / length(diff))
end
