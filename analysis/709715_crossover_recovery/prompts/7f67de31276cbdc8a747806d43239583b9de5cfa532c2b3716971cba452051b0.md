```julia
"""
    symmetric_motif_duplication_gen9_7_gen22_3(tree::N, options, nfeatures::Int, rng::AbstractRNG) where {T,N<:AbstractExpressionNode{T}}

A structural mutation that duplicates a subexpression motif, optionally cyclically shifts its
feature indices, couples it with a target (or a normalizing constant) via a binary operator,
and occasionally wraps the result in a unary operator.

# Core Idea
Physical laws often repeat functional motifs across variables (e.g. `vx^2 + vy^2`) *and* also
normalize such motifs through a surrounding unary function, frequently against unity
(e.g. `sqrt(1 - v^2/c^2)`, `1/(1+exp(-x))`). The parent operator only ever couples the motif with
a copy of an existing tree node via a binary operator, which makes "constant-anchored" normalized
forms (motif vs. `1`) structurally hard to reach, and it never introduces the enclosing unary
wrapper (`sqrt`, `exp`, `log`, ...) that is characteristic of these physical relations.

# Steps
1. Return early if no binary operators are configured.
2. Sample a donor subtree and deep-copy it into an independent motif.
3. If `nfeatures > 1`, cyclically shift the motif's variable leaves by a random offset
   (0 preserves features, enabling self-coupling like `x^2 - x`).
4. Choose a coupling partner: with probability `p_const` use a constant leaf (usually `1.0`,
   biasing toward normalized forms), otherwise copy an existing target node from the tree.
5. Couple the motif and partner under a random binary operator with randomized operand order,
   and splice the result back in place of the target (or, if a constant partner was used, in
   place of the donor's original sibling context via the sampled target node).
6. With probability `p_wrap`, wrap the newly formed node in a random unary operator, directly
   enabling normalization patterns such as `sqrt(1 - motif)` or `exp(motif)`.

# Changes vs. Parent
- Added a constant-anchored coupling path (motif vs. a constant, usually `1.0`) to make
  normalized/relativistic-style expressions structurally reachable.
- Added optional unary-operator wrapping of the coupled node to directly synthesize the
  enclosing function (e.g. `sqrt`) that so often surrounds these motifs in real physics formulae.
- Kept the cyclic feature-shift mechanism and lightweight, unified code path from the parent.
"""
function symmetric_motif_duplication_gen9_7_gen22_3(
    tree::N,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # Guard: binary coupling requires at least one binary operator
    options.nops[2] == 0 && return tree

    # Helper: cyclically permute feature indices in the motif leaves
    function _shift_features!(node, shift::Int, num_features::Int)
        if node.degree == 0
            if !node.constant
                node.feature = mod1(node.feature + shift, num_features)
            end
        else
            for i in 1:(node.degree)
                _shift_features!(get_child(node, i), shift, num_features)
            end
        end
        return node
    end

    # Step 1: Sample a donor subtree motif and duplicate it
    donor = rand(rng, NodeSampler(; tree))
    motif = copy(donor)

    # Step 2: Apply cyclic feature transformation (shift == 0 preserves features for self-coupling)
    if nfeatures > 1
        shift = rand(rng, 0:(nfeatures - 1))
        if shift > 0
            _shift_features!(motif, shift, nfeatures)
        end
    end

    # Step 3: Select the target node to be replaced in the tree
    target = rand(rng, NodeSampler(; tree))

    # Step 4: Choose the coupling partner — either a normalizing constant or a copy of the target.
    # Constant-anchored coupling (bias toward 1.0) directly enables forms like `1 - motif`.
    p_const = 0.25
    if rand(rng) < p_const
        const_val = rand(rng, Bool) ? one(T) : randn(rng, T)
        partner = constructorof(N)(T; val=const_val)
    else
        partner = copy(target)
    end
    bin_op = rand(rng, 1:options.nops[2])

    # Randomize operand ordering to support non-commutative operators (e.g. `1 - motif` vs `motif - 1`)
    children = rand(rng, Bool) ? (partner, motif) : (motif, partner)
    new_node = constructorof(N)(;
        op=bin_op,
        children=children,
    )

    # Step 5: Optionally wrap the coupled node in a unary operator to synthesize the enclosing
    # normalization function (e.g. sqrt(1 - motif)) directly, rather than relying on later mutations.
    p_wrap = 0.3
    if options.nops[1] > 0 && rand(rng) < p_wrap
        una_op = rand(rng, 1:options.nops[1])
        new_node = constructorof(N)(; op=una_op, children=(new_node,))
    end

    # Step 6: Splice the new structure in place of the sampled target node
    set_node!(target, new_node)

    return tree
end
```

## Parent mutation operator 2
```julia
"""
    symmetric_motif_duplication_gen10_1_gen11_9(tree::N, options, nfeatures::Int, rng::AbstractRNG) where {T,N<:AbstractExpressionNode{T}}

A structural mutation operator that duplicates an evolved subexpression *motif*, optionally
re-indexes its variables (cyclic shift or targeted substitution), and couples the copy back into
the tree with a binary operator.

# Core Idea
Ground-truth physical/mathematical laws are frequently built from repeated functional forms applied
to different variables or dimensions (`0.5*m*(vx^2 + vy^2)`, `(x1-x2)^2 + (y1-y2)^2`, `a*b + c*d`).
Random point mutations must rediscover each copy of such a motif independently, which is
exponentially unlikely. This operator instead *reuses* structure that evolution already found:
it clones a subtree, remaps its feature indices, and couples it to a target node.

# Steps
1. Return early if no binary operator exists, or if there is no size budget left.
2. Compute a size budget from `options.maxsize` (resulting size is `size(tree) + size(motif) + 1`),
   and only consider donors that fit.
3. Sample a donor with a tiered preference: (a) compound (`degree > 0`) subtrees containing a
   variable, (b) any subtree containing a variable, (c) any subtree that fits. Compound,
   variable-bearing motifs are the ones for which symmetry duplication is meaningful.
4. Transform the motif's feature indices with one of three modes:
   * cyclic shift by `k ∈ 1:(nfeatures-1)` — replicates a form across a whole variable block,
   * targeted substitution — replaces *one* feature present in the motif everywhere by another,
     which is what pairwise swaps (e.g. `x → y`) require and a global shift cannot express,
   * identity — for self-couplings such as powers (`u*u`) and rational forms (`u/u'`).
5. Couple a copy of a randomly chosen target node with the transformed motif under a binary
   operator, biased toward `+`/`*` (the canonical symmetric accumulators), with randomized
   operand order for non-commutative operators.

# Improvements vs. parent (`symmetric_motif_duplication_gen9_6`)
- **Size budgeting:** the parent could nearly double the tree in one step (donor and target are both
  duplicated), producing offspring that are immediately rejected by the complexity limit. This
  version derives a budget from `options.maxsize` and restricts donor sampling to subtrees that fit,
  so the mutation is far more likely to yield an admissible child.
- **Motif quality filter:** the parent frequently cloned a bare constant/variable leaf, degenerating
  into `insert_random_op`. Tiered sampling now prefers compound, variable-bearing motifs.
- **Richer (but still cheap) feature remapping:** re-adds targeted substitution alongside the cyclic
  shift, recovering pairwise variable exchanges that a single global shift cannot represent, using
  one compact traversal helper instead of the parent's multi-way branch logic.
- **Degeneracy avoidance:** when the donor *is* the target and the identity map is drawn, a nonzero
  shift is forced, avoiding trivially simplifiable `u + u` / `u * u`-only outcomes.
- **Operator bias:** coupling now favors `+` and `*`, matching how symmetric motifs actually combine
  in real laws, instead of picking uniformly among all binary operators.
"""
function symmetric_motif_duplication_gen10_1_gen11_9(
    tree::N,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # Guard: binary coupling requires at least one binary operator
    options.nops[2] == 0 && return tree

    # ---------------------------------------------------------------- size budget
    tree_size = count_nodes(tree)
    maxsize = hasproperty(options, :maxsize) ? options.maxsize : typemax(Int)
    # Resulting size == tree_size + motif_size + 1 (target subtree is duplicated in place)
    budget = maxsize - tree_size - 1
    budget < 1 && return tree

    # ---------------------------------------------------------------- helpers
    # Collect the distinct feature indices appearing in a subtree
    function _collect_features!(acc::Vector{Int}, node)
        if node.degree == 0
            if !node.constant && !(node.feature in acc)
                push!(acc, node.feature)
            end
        else
            for i in 1:(node.degree)
                _collect_features!(acc, get_child(node, i))
            end
        end
        return acc
    end

    # Apply an arbitrary index map to every variable leaf of a subtree
    function _map_features!(node, fmap)
        if node.degree == 0
            if !node.constant
                node.feature = fmap(node.feature)
            end
        else
            for i in 1:(node.degree)
                _map_features!(get_child(node, i), fmap)
            end
        end
        return node
    end

    _fits(t) = count_nodes(t) <= budget
    _has_var(t) = any(n -> n.degree == 0 && !n.constant, t)

    # ---------------------------------------------------------------- Step 1: donor
    # Tier 1: compound, variable-bearing motifs (the interesting symmetry carriers)
    n_tier1 = count(t -> t.degree > 0 && _fits(t) && _has_var(t), tree)
    # Tier 2: any variable-bearing motif that fits
    n_tier2 = count(t -> _fits(t) && _has_var(t), tree)
    n_tier3 = count(_fits, tree)

    donor = if n_tier1 > 0 && rand(rng) < 0.75
        rand(rng, NodeSampler(; tree, filter=t -> t.degree > 0 && _fits(t) && _has_var(t)))
    elseif n_tier2 > 0
        rand(rng, NodeSampler(; tree, filter=t -> _fits(t) && _has_var(t)))
    elseif n_tier3 > 0
        rand(rng, NodeSampler(; tree, filter=_fits))
    else
        return tree  # nothing fits the budget
    end

    motif = copy(donor)

    # ---------------------------------------------------------------- Step 2: target
    target = rand(rng, NodeSampler(; tree))
    self_coupling = target === donor

    # ---------------------------------------------------------------- Step 3: remap features
    if nfeatures > 1
        r = rand(rng)
        if r < 0.45
            # (a) cyclic shift: replicate the form across a block of variables
            shift = rand(rng, 1:(nfeatures - 1))
            _map_features!(motif, f -> mod1(f + shift, nfeatures))
        elseif r < 0.80
            # (b) targeted substitution: swap out exactly one feature of the motif
            feats = _collect_features!(Int[], motif)
            if isempty(feats)
                # constant-only motif: nothing to remap
            else
                src = feats[rand(rng, 1:length(feats))]
                dst = rand(rng, 1:(nfeatures - 1))
                dst >= src && (dst += 1)  # uniform over targets != src
                _map_features!(motif, f -> (f == src ? dst : f))
            end
        else
            # (c) identity map: supports powers / rational self-couplings.
            # Avoid the fully degenerate `u ⊕ u` when donor and target coincide.
            if self_coupling && rand(rng, Bool)
                shift = rand(rng, 1:(nfeatures - 1))
                _map_features!(motif, f -> mod1(f + shift, nfeatures))
            end
        end
    end

    # ---------------------------------------------------------------- Step 4: couple
    # Bias toward the canonical symmetric accumulators `+` and `*`
    binops = options.operators.binops
    pref = Int[]
    for i in 1:(options.nops[2])
        op = binops[i]
        (op === (+) || op === (*)) && push!(pref, i)
    end
    bin_op = if !isempty(pref) && rand(rng) < 0.65
        pref[rand(rng, 1:length(pref))]
    else
        rand(rng, 1:(options.nops[2]))
    end

    target_copy = copy(target)
    # Randomize operand ordering to support non-commutative operators
    children = rand(rng, Bool) ? (target_copy, motif) : (motif, target_copy)
    new_node = constructorof(N)(; op=bin_op, children=children)
    set_node!(target, new_node)

    return tree
end
```
