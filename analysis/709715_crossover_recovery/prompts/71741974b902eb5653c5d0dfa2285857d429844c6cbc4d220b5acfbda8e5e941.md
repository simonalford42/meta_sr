```julia
"""
    relative_aware_shape_log_nmse_loss_gen4_1_gen4_2(tree, dataset, options)

Scale-free fitness that blends three complementary, cheap signals:

1. **Energy-normalized, log-compressed MSE** (`log1p(SSE / Σy²)`) — the dominant
   fidelity term. Normalization makes the loss comparable across differently
   scaled tasks, and the log compression prevents a handful of numerically awful
   candidates from flattening all structural differences.
2. **Soft symmetric relative error** — `mean( r² / (y² + p² + floor) )`, with
   `floor = 1e-3 * mean(y²)`. This is the key addition versus the parent. Plain
   (even normalized) MSE is dominated by the largest-magnitude samples, so on
   multiplicative / power-law targets (most Feynman-style equations, e.g.
   `½m(u²+v²+w²)`, whose targets span two orders of magnitude) a candidate that
   captures the correct *multiplicative* structure on the bulk of the data is
   scored no better than an additive patchwork that happens to nail the few
   high-energy points. The per-point normalizer equalizes the influence of small
   and large targets (an approximately log-space fit), which rewards the correct
   factorized skeleton. It is bounded in [0, 2], hence intrinsically robust to
   outliers, and the additive `floor` (a small fraction of the mean square of the
   targets) keeps points with `y ≈ 0` from saturating the term, so the loss
   remains sane for sign-changing targets.
3. **Affine-invariant shape term** `1 - r²` — retained from the parent so that
   promising nonlinear skeletons survive before their constants/offsets have been
   tuned.

All three terms vanish exactly at a perfect fit, are non-negative, and none of
them looks at expression size (parsimony is handled by `loss_to_cost`).

Implementation notes / changes vs. the parent:
* Target statistics (`mean_y`, `Σ(y-ȳ)²`, `Σy²`) are gathered in a separate,
  prediction-free pass. This is a couple of flops per point (far cheaper than
  the tree evaluation itself) and it lets the second pass use the exact target
  mean, so the covariance is accumulated as `Σ(yᵢ-ȳ)·pᵢ` — algebraically
  identical to the centered covariance but with one fewer running update.
* The relative term needs the target scale, which the pre-pass supplies for free.
* Overflow-safe: infinite/NaN inputs bail out with `Inf`; if the per-point
  relative denominator or numerator overflows, its contribution is clamped to
  its limiting value of one instead of producing `NaN`.
* Exact-zero SSE short-circuits to `0` before any moment sums are inspected.
* Allocation-free (two `@inbounds` loops, `muladd` throughout).
"""
function relative_aware_shape_log_nmse_loss_gen4_1_gen4_2(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    y_vec = dataset.y
    n = length(y_vec)
    if n == 0 || length(prediction) != n
        return L(Inf)
    end

    # ---------------------------------------------------------------------
    # Pass A: target-only moments (Welford for stability) plus target energy.
    # Independent of the candidate, but cheap relative to tree evaluation.
    # ---------------------------------------------------------------------
    mean_y = zero(L)
    m2_y = zero(L)
    energy_y = zero(L)
    @inbounds for i in 1:n
        y = L(y_vec[i])
        if !isfinite(y)
            return L(Inf)
        end
        energy_y = muladd(y, y, energy_y)
        delta_y = y - mean_y
        mean_y += delta_y / L(i)
        m2_y = muladd(delta_y, y - mean_y, m2_y)
    end
    if !isfinite(energy_y) || !isfinite(m2_y)
        return L(Inf)
    end

    # Soft floor for the relative denominator: keeps near-zero targets from
    # saturating the relative term while leaving large targets untouched.
    rel_floor = L(1e-3) * (energy_y / L(n))

    # ---------------------------------------------------------------------
    # Pass B: prediction-dependent accumulators, single sweep.
    # ---------------------------------------------------------------------
    squared_error = zero(L)
    relative_sum = zero(L)
    mean_p = zero(L)
    m2_p = zero(L)
    cov_sum = zero(L)   # == Σ (yᵢ - ȳ)(pᵢ - p̄) because Σ(yᵢ - ȳ) = 0

    @inbounds for i in 1:n
        y = L(y_vec[i])
        p = L(prediction[i])
        if !isfinite(p)
            return L(Inf)
        end

        residual = p - y
        sq_residual = residual * residual
        squared_error = muladd(residual, residual, squared_error)

        # Soft symmetric relative error: scale-free, bounded in [0, 2].
        denom = muladd(y, y, muladd(p, p, rel_floor))
        if isfinite(denom) && isfinite(sq_residual) && denom > zero(L)
            relative_sum += sq_residual / denom
        else
            # Overflow regime: the ratio tends to ~1; clamp instead of NaN.
            relative_sum += one(L)
        end

        # Prediction variance (Welford) and covariance against fixed target mean.
        delta_p = p - mean_p
        mean_p += delta_p / L(i)
        m2_p = muladd(delta_p, p - mean_p, m2_p)
        cov_sum = muladd(y - mean_y, p, cov_sum)
    end

    # Preserve an exact zero optimum regardless of other sums.
    if squared_error == zero(L)
        return zero(L)
    end

    if !isfinite(squared_error) || !isfinite(m2_p) || !isfinite(cov_sum) ||
       !isfinite(relative_sum)
        return L(Inf)
    end

    # ---------------------------------------------------------------------
    # Term 1: energy-normalized, log-compressed MSE.
    # ---------------------------------------------------------------------
    normalizer = energy_y > zero(L) ? energy_y : L(n)
    normalized_mse = squared_error / normalizer
    if !isfinite(normalized_mse)
        return L(Inf)
    end

    # Term 2: mean soft relative error (already bounded by construction).
    relative_loss = relative_sum / L(n)

    # ---------------------------------------------------------------------
    # Term 3: affine-invariant shape mismatch, 1 - r².
    # ---------------------------------------------------------------------
    shape_loss = zero(L)
    if m2_y > zero(L)
        shape_loss = one(L)   # constant predictions explain no target variance
        if m2_p > zero(L)
            correlation = cov_sum / sqrt(m2_p) / sqrt(m2_y)
            if !isfinite(correlation)
                return L(Inf)
            end
            bounded_correlation = min(abs(correlation), one(L))
            shape_loss = one(L) - bounded_correlation * bounded_correlation
        end
    end

    # Direct fidelity stays dominant; the relative term equalizes magnitudes and
    # the shape term supplies pre-constant-tuning guidance.
    loss_value = L(0.55) * log1p(normalized_mse) +
                 L(0.30) * relative_loss +
                 L(0.15) * shape_loss
    return isfinite(loss_value) ? L(loss_value) : L(Inf)
end
```

## Parent loss operator 2
```julia
"""
    shape_guided_log_nmse_loss_gen3_3(tree, dataset, options)

Evaluate an expression with a scale-normalized MSE augmented by a lightweight
shape-matching signal. The direct error is normalized by target energy and
log-compressed so numerically poor candidates do not overwhelm useful structural
differences. A secondary `1 - r²` term measures agreement after the best implicit
affine rescaling, helping promising nonlinear structure survive before its
constants and offset are fully optimized.

Compared with the parent MSE, this loss is scale-independent, less dominated by
large aggregate errors, allocation-free, and provides affine-invariant shape
guidance. Online moments compute all statistics in one pass. Constant targets
disable the shape term, exact predictions return zero, and no expression
complexity penalty is included.
"""
function shape_guided_log_nmse_loss_gen3_3(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = length(dataset.y)
    if n == 0 || length(prediction) != n
        return L(Inf)
    end

    # Welford accumulators provide stable variances and covariance in one pass.
    mean_y = zero(L)
    mean_p = zero(L)
    m2_y = zero(L)
    m2_p = zero(L)
    covariance = zero(L)
    target_energy = zero(L)
    squared_error = zero(L)

    @inbounds for i in 1:n
        y = L(dataset.y[i])
        p = L(prediction[i])
        if !isfinite(y) || !isfinite(p)
            return L(Inf)
        end

        residual = p - y
        squared_error = muladd(residual, residual, squared_error)
        target_energy = muladd(y, y, target_energy)

        count = L(i)
        delta_y = y - mean_y
        delta_p = p - mean_p
        mean_y += delta_y / count
        mean_p += delta_p / count

        m2_y = muladd(delta_y, y - mean_y, m2_y)
        m2_p = muladd(delta_p, p - mean_p, m2_p)
        covariance = muladd(delta_p, y - mean_y, covariance)
    end

    # Preserve an exact zero optimum even if unrelated moment sums overflow.
    if squared_error == zero(L)
        return zero(L)
    end

    if !isfinite(squared_error) || !isfinite(target_energy) ||
       !isfinite(m2_y) || !isfinite(m2_p) || !isfinite(covariance)
        return L(Inf)
    end

    # Normalization preserves relative accuracy across differently scaled tasks.
    normalizer = target_energy > zero(L) ? target_energy : L(n)
    normalized_mse = squared_error / normalizer
    if !isfinite(normalized_mse)
        return L(Inf)
    end

    # This equals the normalized residual of the best affine fit a*p + b.
    shape_loss = zero(L)
    if m2_y > zero(L)
        shape_loss = one(L)
        if m2_p > zero(L)
            correlation = covariance / sqrt(m2_p) / sqrt(m2_y)
            if !isfinite(correlation)
                return L(Inf)
            end
            bounded_correlation = min(abs(correlation), one(L))
            shape_loss = one(L) - bounded_correlation * bounded_correlation
        end
    end

    # Direct fidelity remains dominant; the shape term supplies early guidance.
    loss_value = L(0.85) * log1p(normalized_mse) + L(0.15) * shape_loss
    return isfinite(loss_value) ? L(loss_value) : L(Inf)
end
```
