```julia
# Custom Mutation: add_constant_offset
# =====================================
# This mutation selects a random subtree and wraps it with an addition
# of a random constant: `subtree` -> `subtree + c`
#
# This is different from built-in mutations:
# - mutate_constant: only perturbs EXISTING constants
# - add_node: adds operators at LEAF nodes only
# - insert_node: inserts operator but uses random leaves, not the subtree
#
# This mutation introduces a new constant offset to any part of the tree,
# which can help discover formulas with additive terms.

# Note: This file is `include`d into CustomMutationsModule, so it has access
# to: AbstractExpressionNode, NodeSampler, constructorof, set_node!, etc.

"""
    add_constant_offset(tree, dataset, options, nfeatures, rng)

Wrap a random subtree with addition of a random constant.
`subtree` becomes `subtree + c` where `c` is sampled from normal distribution.

`dataset` is accepted for signature compatibility with data-aware mutations
but is unused here.
"""
function add_constant_offset(
    tree::N,
    dataset,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # Find the + operator index (binary operators are indexed by their position)
    plus_idx = findfirst(op -> op == (+), options.operators.binops)

    if plus_idx === nothing
        # No + operator available, return tree unchanged
        return tree
    end

    # Sample a random node to wrap
    node = rand(rng, NodeSampler(; tree))

    # Create a random constant
    constant_value = randn(rng, T)  # Sample from normal distribution
    constant_node = constructorof(N)(T; val=constant_value)

    # Create new node: node + constant
    # Randomly decide if constant goes on left or right
    if rand(rng, Bool)
        new_node = constructorof(N)(; op=plus_idx, children=(copy(node), constant_node))
    else
        new_node = constructorof(N)(; op=plus_idx, children=(constant_node, copy(node)))
    end

    # Replace the selected node with the wrapped version
    set_node!(node, new_node)

    return tree
end
```

## Parent mutation operator 2
```julia
"""
    fit_rational_pole_mutation_gen4_3(tree, dataset, options, nfeatures, rng)

A data-aware mutation that converts a *polynomial-looking* expression into a
**rational (pole-bearing) expression** by fitting the denominator with least squares.

Motivation: plain genetic search very often approximates a rational law such as
`y = N / (1 - u/3)` by its Taylor series (`N*(1 + u/3 + u^2/9 + ...)`), which
produces long, ever-growing polynomial trees that never cross over to the compact
rational form (this is visible on tasks such as `feynman_II_11_27`). This mutation
provides that jump in one step. Given the current tree `p(x)` (interpreted as a
candidate *numerator*) it searches for a "pole driver" subexpression `q(x)` and
proposes

    p(x)  ->  a * p(x) / (1 + b * q(x)),

where the pair `(a, b)` is obtained in closed form: the condition
`y * (1 + b*q) ≈ a * p` is **linear** in `(a, b)`, so a 2x2 weighted normal-equation
solve yields the optimal coefficients without any nonlinear optimization.

Steps: (1) evaluate the tree on `dataset.X`; (2) collect candidate `q`'s — a random
handful of subtrees of the tree, plus a random feature leaf and a random
product-of-two-features (cheap way to discover denominators like `x0*x1` that are
not yet in the tree); (3) for each candidate solve the 2x2 least-squares system for
`(a, b)`; (4) reject candidates whose fitted denominator `1 + b*q` changes sign or
comes close to zero anywhere on the training data (that would place a pole inside
the data range), or whose `b` is numerically negligible; (5) score surviving
candidates by their *actual* weighted MSE and accept the best one only if it
improves the current MSE by at least 10% (the wrap costs ~4-6 nodes, so we demand a
real gain) and respects `options.maxsize`. Otherwise the tree is returned unchanged.

Assumptions/heuristics: requires `+`, `*`, `/` among the binary operators and a
single-output dataset; constants `a`, `b` are inserted as ordinary constant leaves
so downstream constant optimization can refine them further.
"""
function fit_rational_pole_mutation_gen4_3(
    tree::N,
    dataset,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    y = dataset.y
    y === nothing && return tree            # multi-output: nothing to fit against
    X = dataset.X
    n = dataset.n
    n < 8 && return tree                    # too few points to trust a 2-parameter fit

    # We need +, * and / to be able to express a*p/(1 + b*q)
    binops = options.operators.binops
    i_add = findfirst(op -> op === (+), binops)
    i_mul = findfirst(op -> op === (*), binops)
    i_div = findfirst(op -> op === (/), binops)
    (i_add === nothing || i_mul === nothing || i_div === nothing) && return tree

    # ---- current predictions (the candidate "numerator") -------------------
    p, ok = eval_tree_array(tree, X, options.operators)
    ok || return tree
    all(isfinite, p) || return tree

    w = dataset.weights
    _wmean(v) = w === nothing ? mean(v) : sum(w .* v) / sum(w)

    base_mse = Float64(_wmean(abs2.(y .- p)))
    (!isfinite(base_mse) || base_mse <= 0) && return tree   # nothing to gain

    # ---- assemble candidate pole-drivers q --------------------------------
    subtrees = N[]
    any(tree) do t
        # keep operator nodes and variable leaves; constants make useless denominators
        if t.degree > 0 || !t.constant
            push!(subtrees, t)
        end
        return false                       # never short-circuit: visit all nodes
    end

    cands = N[]
    if !isempty(subtrees)
        # sample a small random handful (duplicates are harmless, just redundant work)
        for _ in 1:min(length(subtrees), 6)
            push!(cands, subtrees[rand(rng, 1:length(subtrees))])
        end
    end
    # a bare feature, and a random pairwise product: cheap "new" denominators
    push!(cands, constructorof(N)(T; feature=rand(rng, 1:nfeatures)))
    if nfeatures >= 2
        f1 = rand(rng, 1:nfeatures)
        f2 = rand(rng, 1:nfeatures)
        push!(
            cands,
            constructorof(N)(;
                op=i_mul,
                children=(
                    constructorof(N)(T; feature=f1), constructorof(N)(T; feature=f2)
                ),
            ),
        )
    end
    isempty(cands) && return tree

    # ---- evaluate every candidate, keep the best rational wrap ------------
    best_mse = base_mse
    best_q = nothing
    best_a = 1.0
    best_b = 0.0

    for q in cands
        qv, qok = eval_tree_array(q, X, options.operators)
        qok || continue
        all(isfinite, qv) || continue
        sq = Float64(std(qv))
        (!isfinite(sq) || sq <= 1e-12) && continue        # constant q ⇒ no new structure

        # Weighted normal equations for  min_{a,b} || y + b*(y*q) - a*p ||^2
        S11 = 0.0; S12 = 0.0; S22 = 0.0; r1 = 0.0; r2 = 0.0
        @inbounds for k in 1:n
            wk = w === nothing ? 1.0 : Float64(w[k])
            yk = Float64(y[k])
            pk = Float64(p[k])
            uk = yk * Float64(qv[k])       # column for b
            S11 += wk * uk * uk
            S12 += wk * uk * pk
            S22 += wk * pk * pk
            r1 += wk * uk * yk
            r2 += wk * pk * yk
        end
        D = S11 * S22 - S12 * S12
        (!isfinite(D) || abs(D) <= 1e-12 * max(1.0, S11 * S22)) && continue

        b = (S12 * r2 - r1 * S22) / D
        a = (S11 * r2 - r1 * S12) / D
        (!isfinite(a) || !isfinite(b)) && continue
        abs(b) * sq < 1e-6 && continue      # denominator would be ~1: no real change
        abs(a) < 1e-12 && continue

        # Pole safety: 1 + b*q must stay away from zero and keep a fixed sign on data
        dmin = Inf
        dmax = -Inf
        @inbounds for k in 1:n
            dk = 1.0 + b * Float64(qv[k])
            dmin = min(dmin, dk)
            dmax = max(dmax, dk)
        end
        (dmin <= 0.1 && dmax >= -0.1) && continue   # straddles / hugs zero ⇒ reject

        # Score the *actual* candidate prediction (not the linearized objective)
        acc = 0.0
        wsum = 0.0
        bad = false
        @inbounds for k in 1:n
            dk = 1.0 + b * Float64(qv[k])
            predk = a * Float64(p[k]) / dk
            if !isfinite(predk)
                bad = true
                break
            end
            wk = w === nothing ? 1.0 : Float64(w[k])
            resid = Float64(y[k]) - predk
            acc += wk * resid * resid
            wsum += wk
        end
        (bad || wsum <= 0) && continue
        mse = acc / wsum
        if isfinite(mse) && mse < best_mse
            best_mse = mse
            best_q = q
            best_a = a
            best_b = b
        end
    end

    # Demand a substantial gain, since the wrap costs several nodes
    (best_q === nothing || best_mse > 0.9 * base_mse) && return tree

    # ---- build  a * tree / (1 + b * q) ------------------------------------
    q_copy = copy(best_q)                  # q may live inside `tree`: must copy
    b_leaf = constructorof(N)(T; val=T(best_b))
    bq = constructorof(N)(; op=i_mul, children=(b_leaf, q_copy))
    one_leaf = constructorof(N)(T; val=one(T))
    denom = constructorof(N)(; op=i_add, children=(one_leaf, bq))

    numer = if abs(best_a - 1.0) < 1e-3
        tree                                # scale is ~1: skip the extra node
    else
        constructorof(N)(;
            op=i_mul, children=(constructorof(N)(T; val=T(best_a)), tree)
        )
    end

    new_root = constructorof(N)(; op=i_div, children=(numer, denom))

    # Respect the size budget if the options expose one
    if hasproperty(options, :maxsize) && count_nodes(new_root) > options.maxsize
        return tree
    end

    return new_root                        # new root!
end
```
