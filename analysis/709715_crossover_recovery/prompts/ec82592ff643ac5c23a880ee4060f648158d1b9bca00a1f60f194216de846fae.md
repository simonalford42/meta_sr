```julia
"""
    symmetric_motif_duplication_gen4_9(tree::N, options, nfeatures::Int, rng::AbstractRNG) where {T,N<:AbstractExpressionNode{T}}

A structural mutation operator that identifies, replicates, and recombines modular subexpressions
(motifs) with optional feature-space permutations or direct coupling.

# Motivation & Core Idea
Physical and mathematical expressions commonly exhibit symmetries, repeated structural motifs
(e.g., Euclidean distances `(x1 - x2)^2 + (y1 - y2)^2`, coupled harmonic terms `sin(x1) + sin(x2)`,
or rational envelopes `u / (1 + u^2)`), and multilinear interactions. Standard point mutations
require exponentially many lucky mutations to replicate an evolved subexpression across other
variables. This operator extracts an existing subtree, systematically permutes its feature indices
(or keeps them identical for self-interaction), and couples or grafts the transformed motif back
into the expression tree.

# Steps & Heuristics
1. Sample a donor subtree from the existing expression tree.
2. Deep-copy the donor into an independent motif.
3. If features are present and `nfeatures > 1`, apply one of three feature-space transformations:
   - Cyclic feature permutation (e.g., `x_i -> x_{i+k}`), preserving relational sub-structures.
   - Targeted single-feature substitution (`x_a -> x_b`), useful for asymmetric couplings.
   - Identity mapping (no change), enabling polynomial powers (e.g. `u * u`) and rational forms.
4. Integrate the transformed motif into the tree via either:
   - Binary operator coupling: wraps a target node and the motif under a random binary operator.
   - Homologous grafting: replaces a target node in the tree with the evolved motif.
"""
function symmetric_motif_duplication_gen4_9(
    tree::N,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # Local helper: recursively collect all active feature indices in a subtree
    function _collect_features!(node, feats::Vector{Int})
        if node.degree == 0
            if !node.constant
                push!(feats, node.feature)
            end
        else
            for i in 1:(node.degree)
                _collect_features!(get_child(node, i), feats)
            end
        end
        return feats
    end

    # Local helper: apply a cyclic shift to all feature leaves in a subtree
    function _shift_features!(node, shift::Int, num_features::Int)
        if node.degree == 0
            if !node.constant
                node.feature = mod1(node.feature + shift, num_features)
            end
        else
            for i in 1:(node.degree)
                _shift_features!(get_child(node, i), shift, num_features)
            end
        end
        return node
    end

    # Local helper: substitute occurrences of a specific feature index with another
    function _remap_feature!(node, from_f::Int, to_f::Int)
        if node.degree == 0
            if !node.constant && node.feature == from_f
                node.feature = to_f
            end
        else
            for i in 1:(node.degree)
                _remap_feature!(get_child(node, i), from_f, to_f)
            end
        end
        return node
    end

    # Fallback: if no binary operators are configured, wrap with a unary operator if possible
    if options.nops[2] == 0
        if options.nops[1] > 0
            node = rand(rng, NodeSampler(; tree))
            new_node = constructorof(N)(;
                op=rand(rng, 1:options.nops[1]),
                children=(copy(node),),
            )
            set_node!(node, new_node)
        end
        return tree
    end

    # Step 1: Select a donor subtree motif to duplicate
    donor = rand(rng, NodeSampler(; tree))
    donor_clone = copy(donor)

    # Step 2: Extract active features and apply feature-space symmetry transformation
    feats = _collect_features!(donor_clone, Int[])
    if !isempty(feats) && nfeatures > 1
        strategy_roll = rand(rng)
        if strategy_roll < 0.45
            # Cyclic permutation: shifts all variable indices by a fixed offset
            shift = rand(rng, 1:(nfeatures - 1))
            _shift_features!(donor_clone, shift, nfeatures)
        elseif strategy_roll < 0.70
            # Targeted substitution: pick one variable present in the motif and remap it
            f_old = rand(rng, feats)
            valid_targets = [f for f in 1:nfeatures if f != f_old]
            if !isempty(valid_targets)
                f_new = rand(rng, valid_targets)
                _remap_feature!(donor_clone, f_old, f_new)
            end
        else
            # Identity: keep exact features to foster powers (e.g. u*u) or rational self-coupling
        end
    end

    # Step 3: Integrate the transformed motif back into the expression
    total_nodes = count_nodes(tree)
    # Use grafting if the tree has sufficient depth/size, otherwise couple via binary operator
    if total_nodes > 3 && rand(rng) < 0.35
        # Homologous Grafting: replace a randomly chosen node with the evolved motif
        target = rand(rng, NodeSampler(; tree))
        set_node!(target, donor_clone)
    else
        # Binary Coupling: pair target node with the motif under a binary operator
        target = rand(rng, Bool) ? donor : rand(rng, NodeSampler(; tree))
        target_copy = copy(target)
        bin_op = rand(rng, 1:options.nops[2])

        # Randomize operand ordering for non-commutative operations (subtraction, division)
        children_tuple = rand(rng, Bool) ? (target_copy, donor_clone) : (donor_clone, target_copy)
        new_node = constructorof(N)(;
            op=bin_op,
            children=children_tuple,
        )
        set_node!(target, new_node)
    end

    return tree
end
```

## Parent mutation operator 2
```julia
"""
    algebraic_refactor_mutation_init_1_gen1_9(tree, options, nfeatures, rng)

A *semantics-preserving* structural mutation that rewrites the expression tree using
elementary algebraic identities instead of perturbing it randomly. This is an
improved variant of `algebraic_refactor_mutation_init_0`.

Core idea (unchanged): PySR's built-in mutations are local edits, so algebraically
equivalent-but-distant forms (e.g. `x1*x2 + x1*x3` vs `x1*(x2+x3)`) are separated by
many individually-deleterious steps. This operator jumps directly between such forms
via a catalogue of rewrite rules applied at a randomly (weighted) chosen site, so that
loss-preserving complexity reduction/reshaping can happen in one move.

What changed vs. the parent:
1. **Removed the overly-eager early exit.** The parent bailed out entirely unless a
   `*` or `/` operator existed in the operator set, even though several useful rewrites
   (constant folding, `a-a`, additive identities) need only `+`/`-`. All multiplicative
   rules already self-guard via `is_times`/`is_div`/explicit `!== nothing` checks, so
   removing the blanket bail-out is safe and unlocks simplification for `+`,`-`-only
   trees (or trees with only unary operators).
2. **Generic constant folding** (new): any binary node whose two children are both
   constant leaves is folded to a single constant leaf using the *actual* operator
   function (not just `+,-,*,/`), and likewise for unary nodes applied to a constant
   leaf. This lets constant sub-expressions created by other mutations (or by this
   mutation's own rewrites) collapse immediately instead of waiting on `mutate_constant`
   or built-in simplification passes. Folds are skipped if the result is non-finite.
3. **Literal 0/1 identities** (new): `a+0`, `0+a`, `a-0` -> `a`; `a*1`, `1*a` -> `a`;
   `a*0`, `0*a` -> `0`; `a/1` -> `a`. These frequently appear after cancellation-style
   rewrites (which can produce `0`/`1` leaves) or after `mutate_constant` drifts a
   constant to exactly one of these values.
4. **Generalized self-cancellation** (new): any `a - a -> 0` and `a / a -> 1` for
   *structurally equal* subtrees `a` (not only the specific sum/product shapes handled
   by the parent's rule (1)), using the same generic non-zero/non-degenerate
   assumption already relied on by the parent's multiplicative-cancellation rules.
5. **Division distributes over its numerator** (new, symmetric to the parent's
   times-distribution rule): `(b±c)/a -> b/a ± c/a`, gated by the same "duplicated
   subtree must be small and the result must fit `maxsize`" heuristic used for `*`
   distribution, and given the same low sampling weight since it is size-increasing.
6. Constant-fold / identity / self-cancellation rewrites are given the highest sampling
   weight (5.0), since they are pure size reductions with no risk of missing an
   opportunity to further simplify in a later generation.
All other logic (structural equality check, factoring, common-denominator collection,
multiplicative cancellation, division folding/lifting, weighted site sampling) is kept
from the parent.
"""
function algebraic_refactor_mutation_init_1_gen1_9(
    tree::N,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    tree.degree == 0 && return tree

    binops = options.operators.binops
    unaops = options.operators.unaops
    plus_idx = findfirst(op -> op == (+), binops)
    minus_idx = findfirst(op -> op == (-), binops)
    times_idx = findfirst(op -> op == (*), binops)
    div_idx = findfirst(op -> op == (/), binops)

    # NOTE: unlike the parent, we do NOT bail out when `*`/`/` are absent: constant
    # folding, additive identities, and self-cancellation are useful even for
    # `+`/`-`-only (or purely unary) expression trees, and every multiplicative-specific
    # rule below already self-guards via `is_times`/`is_div`/explicit `!== nothing`.

    # ---- structural (syntactic) equality of two subtrees ----
    function trees_equal(a::N, b::N)
        a.degree == b.degree || return false
        if a.degree == 0
            a.constant == b.constant || return false
            return a.constant ? (a.val == b.val) : (a.feature == b.feature)
        end
        a.op == b.op || return false
        for i in 1:(a.degree)
            trees_equal(get_child(a, i), get_child(b, i)) || return false
        end
        return true
    end

    # ---- cheap operator-identity predicates ----
    is_plus(t) = t.degree == 2 && plus_idx !== nothing && t.op == plus_idx
    is_minus(t) = t.degree == 2 && minus_idx !== nothing && t.op == minus_idx
    is_sum(t) = is_plus(t) || is_minus(t)
    is_times(t) = t.degree == 2 && times_idx !== nothing && t.op == times_idx
    is_div(t) = t.degree == 2 && div_idx !== nothing && t.op == div_idx

    # Candidate rewrites: (node to overwrite, replacement subtree) plus a sampling weight.
    sites = Tuple{N,N}[]
    weights = Float64[]
    add_site!(node, repl, w) = (push!(sites, (node, repl)); push!(weights, w); nothing)

    # Build a fresh binary node / constant leaf.
    bin(op, a, b) = constructorof(N)(; op=op, children=(a, b))
    lit(v) = constructorof(N)(T; val=v)

    total = count_nodes(tree)
    maxsize = hasproperty(options, :maxsize) ? options.maxsize : 30

    zeroT = zero(T)
    oneT = one(T)

    # Full traversal: `any` visits every node as long as we keep returning `false`.
    any(tree) do t
        # ---- unary constant folding: f(c) -> literal, for constant leaf c ----
        if t.degree == 1
            child = get_child(t, 1)
            if child.degree == 0 && child.constant
                folded = try
                    unaops[t.op](child.val)
                catch
                    nothing
                end
                if folded !== nothing && isfinite(folded)
                    add_site!(t, lit(T(folded)), 5.0)
                end
            end
            return false
        end

        if t.degree == 2
            c1 = get_child(t, 1)
            c2 = get_child(t, 2)

            # (0a) generic binary constant folding: op(c1, c2) -> literal
            if c1.degree == 0 && c1.constant && c2.degree == 0 && c2.constant
                folded = try
                    binops[t.op](c1.val, c2.val)
                catch
                    nothing
                end
                if folded !== nothing && isfinite(folded)
                    add_site!(t, lit(T(folded)), 5.0)
                end
            end

            # (0b) literal 0/1 identities
            if is_plus(t)
                if c1.degree == 0 && c1.constant && c1.val == zeroT
                    add_site!(t, copy(c2), 5.0)
                elseif c2.degree == 0 && c2.constant && c2.val == zeroT
                    add_site!(t, copy(c1), 5.0)
                end
            elseif is_minus(t)
                if c2.degree == 0 && c2.constant && c2.val == zeroT
                    add_site!(t, copy(c1), 5.0)
                end
            elseif is_times(t)
                if c1.degree == 0 && c1.constant && c1.val == oneT
                    add_site!(t, copy(c2), 5.0)
                elseif c1.degree == 0 && c1.constant && c1.val == zeroT
                    add_site!(t, lit(zeroT), 5.0)
                elseif c2.degree == 0 && c2.constant && c2.val == oneT
                    add_site!(t, copy(c1), 5.0)
                elseif c2.degree == 0 && c2.constant && c2.val == zeroT
                    add_site!(t, lit(zeroT), 5.0)
                end
            elseif is_div(t)
                if c2.degree == 0 && c2.constant && c2.val == oneT
                    add_site!(t, copy(c1), 5.0)
                end
            end

            # (0c) generalized self-cancellation for structurally equal subtrees
            if is_minus(t) && trees_equal(c1, c2)
                add_site!(t, lit(zeroT), 5.0)
            elseif is_div(t) && trees_equal(c1, c2)
                add_site!(t, lit(oneT), 5.0)
            end

            # (1) exact cancellation inside sums: (a+b)-b, (a+b)-a, (a-b)+b, b+(a-b)
            if is_minus(t) && is_plus(c1)
                if trees_equal(get_child(c1, 2), c2)
                    add_site!(t, copy(get_child(c1, 1)), 4.0)
                elseif trees_equal(get_child(c1, 1), c2)
                    add_site!(t, copy(get_child(c1, 2)), 4.0)
                end
            elseif is_plus(t) && is_minus(c1) && trees_equal(get_child(c1, 2), c2)
                add_site!(t, copy(get_child(c1, 1)), 4.0)
            elseif is_plus(t) && is_minus(c2) && trees_equal(get_child(c2, 2), c1)
                add_site!(t, copy(get_child(c2, 1)), 4.0)
            end

            # (2) factor a shared multiplicative term: a*b ± a*c -> a*(b±c)
            if is_sum(t) && is_times(c1) && is_times(c2)
                for i in 1:2, j in 1:2
                    if trees_equal(get_child(c1, i), get_child(c2, j))
                        common = copy(get_child(c1, i))
                        rest1 = copy(get_child(c1, 3 - i))   # the *other* factor of c1
                        rest2 = copy(get_child(c2, 3 - j))   # the *other* factor of c2
                        add_site!(t, bin(times_idx, common, bin(t.op, rest1, rest2)), 3.0)
                        break  # one factoring per node is enough
                    end
                end
            end

            # (3) common denominator: a/c ± b/c -> (a±b)/c
            if is_sum(t) &&
                is_div(c1) &&
                is_div(c2) &&
                trees_equal(get_child(c1, 2), get_child(c2, 2))
                add_site!(
                    t,
                    bin(
                        div_idx,
                        bin(t.op, copy(get_child(c1, 1)), copy(get_child(c2, 1))),
                        copy(get_child(c1, 2)),
                    ),
                    3.0,
                )
            end

            # (4) multiplicative cancellation: (a*b)/a -> b  and  (a*b)/b -> a
            if is_div(t) && is_times(c1)
                if trees_equal(get_child(c1, 1), c2)
                    add_site!(t, copy(get_child(c1, 2)), 4.0)
                elseif trees_equal(get_child(c1, 2), c2)
                    add_site!(t, copy(get_child(c1, 1)), 4.0)
                end
            end

            # (5) multiplicative cancellation: (a/b)*b -> a  and  b*(a/b) -> a
            if is_times(t)
                if is_div(c1) && trees_equal(get_child(c1, 2), c2)
                    add_site!(t, copy(get_child(c1, 1)), 4.0)
                elseif is_div(c2) && trees_equal(get_child(c2, 2), c1)
                    add_site!(t, copy(get_child(c2, 1)), 4.0)
                end
            end

            # (6) fold nested divisions: (a/b)/c -> a/(b*c)
            if is_div(t) && is_div(c1) && times_idx !== nothing
                add_site!(
                    t,
                    bin(
                        div_idx,
                        copy(get_child(c1, 1)),
                        bin(times_idx, copy(get_child(c1, 2)), copy(c2)),
                    ),
                    2.0,
                )
            end

            # (7) lift an inner division: a/(b/c) -> (a*c)/b
            if is_div(t) && is_div(c2) && times_idx !== nothing
                add_site!(
                    t,
                    bin(
                        div_idx,
                        bin(times_idx, copy(c1), copy(get_child(c2, 2))),
                        copy(get_child(c2, 1)),
                    ),
                    2.0,
                )
            end

            # (8) distribution over multiplication (size-increasing): a*(b±c) -> a*b ± a*c
            if is_times(t)
                for k in 1:2
                    s = get_child(t, k)          # candidate sum
                    m = get_child(t, 3 - k)      # multiplier that gets duplicated
                    if is_sum(s)
                        nm = count_nodes(m)
                        # keep the duplicated factor tiny and respect the size budget
                        if nm <= 3 && total + nm + 1 <= maxsize
                            add_site!(
                                t,
                                bin(
                                    s.op,
                                    bin(times_idx, copy(m), copy(get_child(s, 1))),
                                    bin(times_idx, copy(m), copy(get_child(s, 2))),
                                ),
                                1.0,
                            )
                        end
                    end
                end
            end

            # (9) distribution over division (size-increasing): (b±c)/a -> b/a ± c/a
            if is_div(t) && is_sum(c1)
                na = count_nodes(c2)
                if na <= 3 && total + na + 1 <= maxsize
                    add_site!(
                        t,
                        bin(
                            c1.op,
                            bin(div_idx, copy(get_child(c1, 1)), copy(c2)),
                            bin(div_idx, copy(get_child(c1, 2)), copy(c2)),
                        ),
                        1.0,
                    )
                end
            end
        end
        return false  # never short-circuit: we want a complete traversal
    end

    isempty(sites) && return tree  # no identity applies -> leave the tree untouched

    # Weighted sampling: simplifications are much more likely than expansion.
    r = rand(rng) * sum(weights)
    idx = length(weights)
    acc = 0.0
    for (i, w) in enumerate(weights)
        acc += w
        if r <= acc
            idx = i
            break
        end
    end

    node, repl = sites[idx]
    # `repl` was built exclusively from copies, so in-place overwrite is safe
    # (this also works when `node === tree`, so the root never needs special handling).
    set_node!(node, repl)
    return tree
end
```
