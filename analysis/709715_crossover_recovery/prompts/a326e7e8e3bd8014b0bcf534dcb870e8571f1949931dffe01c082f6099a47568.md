## Parent selection operator 1
```julia
"""
    novelty_weighted_quality_gated_niche_tournament_gen18_4(
        pop, running_search_statistics, options
    )

Crowding-aware tournament selection combining conservative lineage suppression,
recency-based turnover, and quality-gated complexity-niche exploration.

The operator synthesizes clone-only crowding with the broader anti-monopoly ideas of
lineage and age diversity:

 1. Sample an enlarged tournament pool and compute the usual
    adaptive-parsimony-adjusted costs.
 2. Process candidates from best to worst. Strongly penalize near-identical members
    having the same complexity and loss, while applying only a small penalty to
    closely related parent/child or sibling candidates when their complexity and
    loss are also similar. This preserves genuinely different descendants.
 3. Give newer members a mild cost discount, attenuated by their crowding weight so
    that fresh structural alternatives are encouraged without rewarding fresh
    clones enough to evade suppression.
 4. Adapt the probability of niche exploration to observed pool crowding. During a
    niche move, first retain the best representative of each valid complexity, then
    choose the rarest complexity only from the better two-thirds of those
    representatives. This repopulates starved complexity bands without selecting an
    arbitrarily poor member merely because its niche is empty.
 5. Otherwise apply the standard geometric tournament ladder to the adjusted scores.

Equal complexity and nearly equal loss are used as a cheap proxy for phenotypic
duplication. Family-based suppression requires additional similarity and is kept
fractional, reflecting that shared ancestry alone does not imply redundancy.
Multiplicative adjustments are sign-safe, so factors above one always worsen a
score and factors below one always improve it.
"""
function novelty_weighted_quality_gated_niche_tournament_gen18_4(
    pop::Population{T,L,N},
    running_search_statistics::RunningSearchStatistics,
    options::AbstractOptions,
)::PopMember{T,L,N} where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    # Mild tunables: clone evidence dominates, while lineage and youth are nudges.
    pool_scale = 1.5
    crowding_penalty = 1.35
    lineage_weight = 0.25
    clone_loss_rtol = 1e-6
    lineage_loss_rtol = 1e-2
    youth_discount = 0.03
    niche_probability_base = 0.08
    niche_probability_boost = 0.10
    niche_probability_max = 0.18

    # Sample a larger pool so local crowding and niche occupancy are observable.
    n_base = options.tournament_selection_n
    pool_size = min(pop.n, max(n_base, round(Int, pool_scale * n_base)))
    pool = StatsBase.sample(pop.members, pool_size; replace=false)
    n = length(pool)

    # Compute complexity and the default adaptive-parsimony-adjusted score.
    complexities = Vector{Int}(undef, n)
    scores = Vector{L}(undef, n)
    adaptive_parsimony_scaling = L(options.adaptive_parsimony_scaling)

    for i in 1:n
        member = pool[i]
        complexity = compute_complexity(member, options)
        complexities[i] = complexity

        if options.use_frequency_in_tournament
            frequency = if 0 < complexity <= options.maxsize
                L(running_search_statistics.normalized_frequencies[complexity])
            else
                zero(L)
            end
            scores[i] =
                member.cost * exp(adaptive_parsimony_scaling * frequency)
        else
            scores[i] = member.cost
        end
    end

    # Better representatives establish the local crowding baseline.
    order = sortperm(scores)
    crowding = zeros(Float64, n)

    for a in 2:n
        i = order[a]
        li = pool[i].loss
        scale_i = max(abs(li), eps(L))
        weight = 0.0

        for b in 1:(a - 1)
            j = order[b]
            lj = pool[j].loss
            loss_scale = max(scale_i, abs(lj))

            same_phenotype =
                complexities[i] == complexities[j] &&
                abs(li - lj) <= L(clone_loss_rtol) * loss_scale

            if same_phenotype
                # Strong evidence that the later member is redundant.
                weight += 1.0
                continue
            end

            parent_i = pool[i].parent
            parent_j = pool[j].parent
            related =
                (parent_i != 0 && parent_i == parent_j) ||
                (parent_i != 0 && parent_i == pool[j].ref) ||
                (parent_j != 0 && parent_j == pool[i].ref)

            # Shared ancestry is penalized only when phenotype is also close.
            if related &&
                abs(complexities[i] - complexities[j]) <= 1 &&
                abs(li - lj) <= L(lineage_loss_rtol) * loss_scale
                weight += lineage_weight
            end
        end

        crowding[i] = weight
    end

    # Find the age range once; recency is used only as a mild novelty preference.
    birth_min = pool[1].birth
    birth_max = pool[1].birth
    for i in 2:n
        birth = pool[i].birth
        birth < birth_min && (birth_min = birth)
        birth > birth_max && (birth_max = birth)
    end
    birth_span = max(birth_max - birth_min, 1)

    penalized = Vector{L}(undef, n)
    crowded_count = 0

    for i in 1:n
        redundancy = crowding[i]
        redundancy > 0.0 && (crowded_count += 1)

        recency = (pool[i].birth - birth_min) / birth_span
        novelty_weight = 1.0 / (1.0 + redundancy)

        # Clone/lineage penalties dominate; youth matters most for uncrowded forms.
        factor =
            crowding_penalty^redundancy *
            (1.0 - youth_discount * recency * novelty_weight)
        typed_factor = L(factor)

        # Sign-safe adjustment: larger factors always make the score less fit.
        penalized[i] = if scores[i] > zero(L)
            scores[i] * typed_factor
        elseif scores[i] < zero(L)
            scores[i] / typed_factor
        else
            scores[i]
        end
    end

    # Increase explicit niche exploration when the sampled pool shows monopoly.
    crowded_fraction = crowded_count / n
    niche_probability = min(
        niche_probability_max,
        niche_probability_base +
        niche_probability_boost * crowded_fraction,
    )

    if rand() < niche_probability
        # Keep one best representative per valid complexity niche.
        penalized_order = sortperm(penalized)
        seen_complexities = falses(options.maxsize)
        niche_champions = Int[]

        for i in penalized_order
            complexity = complexities[i]
            if 0 < complexity <= options.maxsize &&
                !seen_complexities[complexity]
                seen_complexities[complexity] = true
                push!(niche_champions, i)
            end
        end

        if !isempty(niche_champions)
            # Quality gate: consider the better two-thirds of niche champions.
            quality_count = max(1, cld(2 * length(niche_champions), 3))
            best_idx = niche_champions[1]
            best_frequency = Float64(
                running_search_statistics.normalized_frequencies[
                    complexities[best_idx]
                ],
            )
            best_score = penalized[best_idx]

            for position in 2:quality_count
                i = niche_champions[position]
                frequency = Float64(
                    running_search_statistics.normalized_frequencies[
                        complexities[i]
                    ],
                )

                if frequency < best_frequency ||
                    (frequency == best_frequency && penalized[i] < best_score)
                    best_idx = i
                    best_frequency = frequency
                    best_score = penalized[i]
                end
            end

            return pool[best_idx]
        end
    end

    # Standard geometric tournament ladder over diversity-adjusted scores.
    p = options.tournament_selection_p
    chosen_idx = if p == 1.0
        argmin_fast(penalized)
    else
        ranks = collect(0:(n - 1))
        probabilities = p * ((1 - p) .^ ranks)
        weights = StatsBase.Weights(probabilities, sum(probabilities))
        selected_rank = StatsBase.sample(weights)

        if selected_rank == 1
            argmin_fast(penalized)
        else
            bottomk_fast(penalized, selected_rank)[2][end]
        end
    end

    return pool[chosen_idx]
end
```

## Parent selection operator 2
```julia
"""
    plateau_aware_gap_seeking_niche_tournament_gen26_5(
        pop, running_search_statistics, options
    )

Crowding-aware tournament selection that fuses the two parents' clone/lineage
suppression and quality-gated niche exploration with a new *plateau* notion of
redundancy and a *Pareto-gap* notion of where structural innovation is needed.

Motivation (from observed search traces): once a motif such as
`x0 / exp(x3 / (x1 * square(x2)))` takes over, the population spends most of its
budget re-tuning constants of that motif. Every such variant has (nearly) the
same complexity and (nearly) the same loss, so the front creeps down by ~1% per
band while an entirely different structure (e.g. a rational `1/(1 + ...)` form)
is never tried. Two additions target exactly this failure mode:

  * **Plateau crowding.** Beyond exact clones (identical complexity, loss equal to
    within `clone_loss_rtol`) and closely-related kin, we now add a smaller
    penalty for *unrelated* members that merely sit in the same complexity band
    (`|Δcomplexity| <= 1`) with losses within a few percent of a better member.
    These are the constant-tuning twins of an entrenched motif; damping them
    frees tournament slots for genuinely different structures.

  * **Gap-seeking niche moves.** When exploring niches we still keep only the
    best representative per complexity and gate by quality (best two thirds),
    but the winning niche is chosen by a rank-sum of two scale-free criteria:
    (i) rarity of the complexity in `normalized_frequencies`, and (ii) *flatness*
    of the local Pareto front, i.e. the relative loss improvement of that
    champion over the champion of the next smaller occupied complexity. A flat
    step means additional complexity there bought nothing, which is precisely a
    band where structural change, not constant polish, is required.

Steps:

 1. Sample an enlarged pool and compute adaptive-parsimony-adjusted scores.
 2. Walk candidates best-to-worst, accumulating crowding weight from clones
    (full weight), similar kin (fractional), and plateau twins (fractional).
 3. Apply a mild youth discount, attenuated by crowding so that fresh clones
    cannot buy their way out of suppression; adjustments are sign-safe.
 4. With a crowding-adaptive probability, perform the gap-seeking, quality-gated
    niche move described above.
 5. Otherwise run the standard geometric tournament ladder on adjusted scores.

Assumptions: equal complexity plus near-equal loss is a cheap proxy for
phenotypic duplication; shared ancestry or a flat Pareto step are weaker signals
and therefore only nudge the scores rather than dominate them.
"""
function plateau_aware_gap_seeking_niche_tournament_gen26_5(
    pop::Population{T,L,N},
    running_search_statistics::RunningSearchStatistics,
    options::AbstractOptions,
)::PopMember{T,L,N} where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    # Mild tunables: clone evidence dominates; lineage, plateau and youth are nudges.
    pool_scale = 1.5
    crowding_penalty = 1.35
    lineage_weight = 0.25
    plateau_weight = 0.30
    clone_loss_rtol = 1e-6
    lineage_loss_rtol = 1e-2
    plateau_loss_rtol = 3e-2
    youth_discount = 0.03
    niche_probability_base = 0.08
    niche_probability_boost = 0.12
    niche_probability_max = 0.20

    # Sample a larger pool so local crowding and niche occupancy are observable.
    n_base = options.tournament_selection_n
    pool_size = min(pop.n, max(n_base, round(Int, pool_scale * n_base)))
    pool = StatsBase.sample(pop.members, pool_size; replace=false)
    n = length(pool)

    # Complexity and default adaptive-parsimony-adjusted score.
    complexities = Vector{Int}(undef, n)
    scores = Vector{L}(undef, n)
    adaptive_parsimony_scaling = L(options.adaptive_parsimony_scaling)

    for i in 1:n
        member = pool[i]
        complexity = compute_complexity(member, options)
        complexities[i] = complexity

        if options.use_frequency_in_tournament
            frequency = if 0 < complexity <= options.maxsize
                L(running_search_statistics.normalized_frequencies[complexity])
            else
                zero(L)
            end
            scores[i] = member.cost * exp(adaptive_parsimony_scaling * frequency)
        else
            scores[i] = member.cost
        end
    end

    # Better representatives establish the local crowding baseline.
    order = sortperm(scores)
    crowding = zeros(Float64, n)

    for a in 2:n
        i = order[a]
        li = pool[i].loss
        scale_i = max(abs(li), eps(L))
        weight = 0.0

        for b in 1:(a - 1)
            j = order[b]
            lj = pool[j].loss
            loss_scale = max(scale_i, abs(lj))

            same_complexity = complexities[i] == complexities[j]
            loss_gap = abs(li - lj)

            # (1) Exact phenotypic clone: strong evidence of redundancy.
            if same_complexity && loss_gap <= L(clone_loss_rtol) * loss_scale
                weight += 1.0
                continue
            end

            parent_i = pool[i].parent
            parent_j = pool[j].parent
            related =
                (parent_i != 0 && parent_i == parent_j) ||
                (parent_i != 0 && parent_i == pool[j].ref) ||
                (parent_j != 0 && parent_j == pool[i].ref)

            near_band = abs(complexities[i] - complexities[j]) <= 1

            # (2) Shared ancestry, penalized only when phenotype is also close.
            if related && near_band && loss_gap <= L(lineage_loss_rtol) * loss_scale
                weight += lineage_weight
                continue
            end

            # (3) Plateau twin: same band, loss within a few percent of a better
            #     member, no ancestry evidence needed. This is the signature of a
            #     dominant motif being re-tuned rather than genuinely re-designed.
            if near_band && loss_gap <= L(plateau_loss_rtol) * loss_scale
                weight += plateau_weight
            end
        end

        crowding[i] = weight
    end

    # Age range for the mild recency preference.
    birth_min = pool[1].birth
    birth_max = pool[1].birth
    for i in 2:n
        birth = pool[i].birth
        birth < birth_min && (birth_min = birth)
        birth > birth_max && (birth_max = birth)
    end
    birth_span = max(birth_max - birth_min, 1)

    penalized = Vector{L}(undef, n)
    crowded_count = 0

    for i in 1:n
        redundancy = crowding[i]
        redundancy > 0.0 && (crowded_count += 1)

        recency = (pool[i].birth - birth_min) / birth_span
        novelty_weight = 1.0 / (1.0 + redundancy)

        # Crowding penalties dominate; youth helps most for uncrowded forms.
        factor =
            crowding_penalty^redundancy *
            (1.0 - youth_discount * recency * novelty_weight)
        typed_factor = L(factor)

        # Sign-safe: factors above one always worsen, below one always improve.
        penalized[i] = if scores[i] > zero(L)
            scores[i] * typed_factor
        elseif scores[i] < zero(L)
            scores[i] / typed_factor
        else
            scores[i]
        end
    end

    # More explicit niche exploration when the pool looks monopolized.
    crowded_fraction = crowded_count / n
    niche_probability = min(
        niche_probability_max,
        niche_probability_base + niche_probability_boost * crowded_fraction,
    )

    if rand() < niche_probability
        # Best representative per valid complexity niche, in quality order.
        penalized_order = sortperm(penalized)
        seen_complexities = falses(options.maxsize)
        niche_champions = Int[]

        for i in penalized_order
            complexity = complexities[i]
            if 0 < complexity <= options.maxsize && !seen_complexities[complexity]
                seen_complexities[complexity] = true
                push!(niche_champions, i)
            end
        end

        n_champ = length(niche_champions)
        if n_champ > 0
            # Flatness of the local Pareto step: relative loss improvement of each
            # champion over the champion of the next smaller occupied complexity.
            # Small values mark bands where extra complexity bought nothing.
            flatness = ones(Float64, n_champ)
            by_complexity = sortperm(
                collect(1:n_champ); by=k -> complexities[niche_champions[k]]
            )
            for t in 2:n_champ
                k = by_complexity[t]
                kprev = by_complexity[t - 1]
                loss_prev = Float64(pool[niche_champions[kprev]].loss)
                loss_cur = Float64(pool[niche_champions[k]].loss)
                denom = max(abs(loss_prev), 1e-12)
                flatness[k] = max(0.0, (loss_prev - loss_cur) / denom)
            end

            # Quality gate: only the better two thirds of champions compete.
            quality_count = max(1, cld(2 * n_champ, 3))
            candidates = collect(1:quality_count)

            # Rank-sum of rarity and flatness (both "smaller is more attractive"),
            # which keeps the two criteria on a common, scale-free footing.
            freqs = Float64[
                Float64(
                    running_search_statistics.normalized_frequencies[
                        complexities[niche_champions[k]]
                    ],
                ) for k in candidates
            ]
            flats = Float64[flatness[k] for k in candidates]

            freq_ranks = zeros(Int, quality_count)
            flat_ranks = zeros(Int, quality_count)
            for (r, k) in enumerate(sortperm(freqs))
                freq_ranks[k] = r
            end
            for (r, k) in enumerate(sortperm(flats))
                flat_ranks[k] = r
            end

            best_pos = 1
            best_rank_sum = freq_ranks[1] + flat_ranks[1]
            best_score = penalized[niche_champions[1]]
            for k in 2:quality_count
                rank_sum = freq_ranks[k] + flat_ranks[k]
                score_k = penalized[niche_champions[k]]
                # Ties broken by adjusted quality, so we never take a clearly
                # worse member merely because its band looks attractive.
                if rank_sum < best_rank_sum ||
                    (rank_sum == best_rank_sum && score_k < best_score)
                    best_pos = k
                    best_rank_sum = rank_sum
                    best_score = score_k
                end
            end

            return pool[niche_champions[best_pos]]
        end
    end

    # Standard geometric tournament ladder over diversity-adjusted scores.
    p = options.tournament_selection_p
    chosen_idx = if p == 1.0
        argmin_fast(penalized)
    else
        ranks = collect(0:(n - 1))
        probabilities = p * ((1 - p) .^ ranks)
        weights = StatsBase.Weights(probabilities, sum(probabilities))
        selected_rank = StatsBase.sample(weights)

        if selected_rank == 1
            argmin_fast(penalized)
        else
            bottomk_fast(penalized, selected_rank)[2][end]
        end
    end

    return pool[chosen_idx]
end
```
