## Parent loss operator 1

```julia
"""
    affine_tail_continuation_loss_gen3_9(tree, dataset, options)

Variance-normalized error with two complementary correction terms, computed in a single
cheap pass over the data:

1. **Raw fit term** — `log1p(SSE / SS_y)`, the usual normalized-MSE signal.
2. **Tail-concentration term** — `log1p(sqrt(mean(u^4)))` where `u_i = d_i / sd(y)` are the
   normalized residuals. Because `sqrt(mean(u^4)) >= mean(u^2)`, this term only exceeds the
   raw term when the residuals are *concentrated* in a subregion of the data. Smooth
   truncated-series surrogates (e.g. `x0 + x0*x1^2/(2*x2^2)` standing in for
   `x0/sqrt(1-x1^2/x2^2)`) fit most of the domain almost perfectly and dump all of their
   error near the singular/extreme region; the L4 term makes those plateaus visibly worse
   than structurally correct forms, discouraging the epicycle-stacking seen in the parent's
   Pareto fronts. Its weight is kept modest (1/3 in log space) so i.i.d. observation noise
   — whose residuals are spread out, giving `mean(u^4) ~ 3*mean(u^2)^2` — is not
   over-penalized on noisy real-world data.
3. **Affine-calibration scaffold** — the least-squares residual of `a*prediction + b`,
   obtained analytically from a Welford pass (`SS_y - cov(p,y)^2/var(p)`). This rewards
   expressions with the right functional shape before their constants are tuned, and its
   weight `0.75 * nmse / (1 + nmse)` fades continuously to zero as the raw fit improves, so
   only genuinely correct predictions attain zero loss.

Changes vs. the parent (`affine_continuation_loss_init_8`):
* Added the L4 tail-concentration term (main new signal), normalized to be directly
  comparable to the raw NMSE and folded in with a small, fixed log-space weight.
* Replaced the parent's Welford pass over the residuals with direct accumulation of
  `sum d^2` and `sum d^4`; these are sums of non-negative terms, so there is no
  cancellation to protect against, and it removes three updates per sample.
* Added an overflow-safe fallback: if the fourth-power sum is not finite, the tail term
  degrades gracefully to the raw term instead of returning `Inf`.
* Kept the relative variance floors, constant-target/constant-prediction guards, and the
  continuation schedule of the parent unchanged. No complexity penalty is included.
"""
function affine_tail_continuation_loss_gen3_9(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = dataset.n
    if n <= 0 || length(prediction) != n || length(dataset.y) != n
        return L(Inf)
    end

    zero_l = zero(L)
    one_l = one(L)

    # Welford state for the affine calibration (needs centered moments of p and y).
    mean_p = zero_l
    mean_y = zero_l
    m2_p = zero_l
    m2_y = zero_l
    cov_py = zero_l

    # Residual power sums: both are sums of non-negative terms, so plain
    # accumulation is exact enough (no catastrophic cancellation possible).
    sse = zero_l
    s4 = zero_l

    @inbounds for i in 1:n
        p = L(prediction[i])
        y = L(dataset.y[i])
        d = p - y

        if !(isfinite(p) && isfinite(y) && isfinite(d))
            return L(Inf)
        end

        k = L(i)

        delta_p = p - mean_p
        mean_p += delta_p / k

        delta_y = y - mean_y
        mean_y += delta_y / k

        # Covariance update uses the freshly updated target mean (standard form).
        cov_py += delta_p * (y - mean_y)
        m2_p += delta_p * (p - mean_p)
        m2_y += delta_y * (y - mean_y)

        d2 = d * d
        sse += d2
        s4 += d2 * d2
    end

    if !(isfinite(mean_p) && isfinite(mean_y) &&
         isfinite(m2_p) && isfinite(m2_y) && isfinite(cov_py) && isfinite(sse))
        return L(Inf)
    end

    # Centered sums can drift microscopically negative from roundoff.
    m2_p = max(m2_p, zero_l)
    m2_y = max(m2_y, zero_l)
    sse = max(sse, zero_l)

    if sse == zero_l
        return zero_l
    end

    n_l = L(n)

    # Relative floors suppress unstable calibration on numerically constant data.
    y_reference = max(one_l, abs(mean_y))
    p_reference = max(one_l, abs(mean_p))
    scale_floor = n_l * eps(L) * y_reference * y_reference
    prediction_floor = n_l * eps(L) * p_reference * p_reference
    if !(isfinite(scale_floor) && isfinite(prediction_floor))
        return L(Inf)
    end

    scale_sse = max(m2_y, scale_floor)
    target_is_variable = m2_y > scale_floor

    # --- affine-calibrated structural error -------------------------------------
    affine_sse = sse
    if target_is_variable
        if m2_p > prediction_floor
            # Explained SSE of the least-squares fit a*p+b is cov(p,y)^2 / var(p).
            normalized_covariance = cov_py / sqrt(m2_p)
            explained_sse = normalized_covariance * normalized_covariance
            if !isfinite(explained_sse)
                return L(Inf)
            end
            affine_sse = clamp(m2_y - explained_sse, zero_l, m2_y)
        else
            # A constant prediction can only capture the target intercept.
            affine_sse = m2_y
        end
    end

    raw_nmse = sse / scale_sse
    affine_nmse = affine_sse / scale_sse
    if !(isfinite(raw_nmse) && isfinite(affine_nmse))
        return L(Inf)
    end

    # --- tail-concentration term -------------------------------------------------
    # With u_i = d_i / sd(y): mean(u^2) == raw_nmse and
    # sqrt(mean(u^4)) == sqrt(n * s4) / scale_sse (>= raw_nmse, equal iff |d| constant).
    tail_nmse = raw_nmse
    if isfinite(s4) && s4 > zero_l
        candidate = sqrt(n_l * s4) / scale_sse
        # Fall back to the raw term on overflow; clamp guards roundoff below raw.
        tail_nmse = isfinite(candidate) ? max(candidate, raw_nmse) : raw_nmse
    end

    raw_term = log1p(max(raw_nmse, zero_l))
    tail_term = log1p(max(tail_nmse, zero_l))
    affine_term = log1p(max(affine_nmse, zero_l))

    # Modest, fixed weight on residual concentration: enough to break ties between
    # globally-smooth-but-locally-wrong surrogates and correct structure, small
    # enough that spread-out observation noise is not over-penalized.
    tail_weight = L(1) / L(3)
    fit_term = (one_l - tail_weight) * raw_term + tail_weight * tail_term

    # Structural (affine) guidance is strong far from the target and vanishes near a fit,
    # so only exact predictions can reach zero loss.
    scaffold_strength = L(3) / L(4)
    blend = scaffold_strength * raw_nmse / (one_l + raw_nmse)
    loss_value = (one_l - blend) * fit_term + blend * affine_term

    return isfinite(loss_value) ? L(max(loss_value, zero_l)) : L(Inf)
end
```

## Parent loss operator 2
```julia
"""
    adaptive_shape_continuation_nmse_loss_gen8_3(tree, dataset, options)

Combine the *affine continuation* idea of parent 1 with the *bounded shape signal*
of parent 2 into a single-pass, allocation-free loss.

Core idea:
  * One numerically stable Welford pass accumulates the raw sum of squared
    residuals together with the first/second moments of the prediction and the
    target and their covariance, plus the raw target energy as a fallback scale.
  * Fidelity term: the raw SSE is normalized by the target variance (falling back
    to target energy, then to `n`, for numerically constant targets) and
    compressed with `log1p`, so catastrophic candidates cannot drown out
    structurally useful ones while relative accuracy stays scale-free.
  * Shape term: `1 - r²`, i.e. the *bounded* normalized residual left after the
    best implicit affine map `a * prediction + b`. This rewards candidates whose
    functional shape is already correct even though their constants/offset have
    not been tuned yet. Being bounded in `[0, 1]`, it can never dominate the
    ranking the way an unbounded calibrated error can.
  * Continuation schedule: the weight on the shape term is not fixed but grows
    with the raw error, `w = w_max * raw_nmse / (1 + raw_nmse)`, so structural
    guidance is strong far from a fit and fades smoothly to zero as the raw
    error vanishes. Only genuinely correct predictions reach zero loss, avoiding
    the degenerate optimum of a purely affine-invariant objective.

Heuristics/assumptions: `w_max = 0.5` keeps direct fidelity at least half the
loss everywhere; constant targets disable the shape term (correlation is
undefined); a constant prediction against a variable target gets the maximal
shape penalty of 1. No expression-complexity penalty is included.
"""
function adaptive_shape_continuation_nmse_loss_gen8_3(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = dataset.n
    if n <= 0 || length(prediction) != n || length(dataset.y) != n
        return L(Inf)
    end

    zero_l = zero(L)
    one_l = one(L)

    # Welford accumulators: stable variances/covariance without raw power sums.
    mean_y = zero_l
    mean_p = zero_l
    m2_y = zero_l
    m2_p = zero_l
    covariance = zero_l
    squared_error = zero_l
    target_energy = zero_l

    @inbounds for i in 1:n
        y = L(dataset.y[i])
        p = L(prediction[i])
        if !(isfinite(y) && isfinite(p))
            return L(Inf)
        end

        residual = p - y
        squared_error = muladd(residual, residual, squared_error)
        target_energy = muladd(y, y, target_energy)

        count = L(i)
        delta_y = y - mean_y
        delta_p = p - mean_p
        mean_y += delta_y / count
        mean_p += delta_p / count

        # Updates use the freshly advanced means (standard Welford form).
        m2_y = muladd(delta_y, y - mean_y, m2_y)
        m2_p = muladd(delta_p, p - mean_p, m2_p)
        covariance = muladd(delta_p, y - mean_y, covariance)
    end

    # Preserve the exact zero optimum even if auxiliary sums are extreme.
    if squared_error == zero_l
        return zero_l
    end

    if !(isfinite(squared_error) && isfinite(target_energy) &&
         isfinite(m2_y) && isfinite(m2_p) && isfinite(covariance) &&
         isfinite(mean_y) && isfinite(mean_p))
        return L(Inf)
    end

    # Roundoff can make centered sums microscopically negative.
    m2_y = max(m2_y, zero_l)
    m2_p = max(m2_p, zero_l)

    # Relative floors keep us from "calibrating" numerically constant data.
    n_l = L(n)
    y_reference = max(one_l, abs(mean_y))
    p_reference = max(one_l, abs(mean_p))
    target_floor = n_l * eps(L) * y_reference * y_reference
    prediction_floor = n_l * eps(L) * p_reference * p_reference
    if !(isfinite(target_floor) && isfinite(prediction_floor))
        return L(Inf)
    end

    target_is_variable = m2_y > target_floor

    # Scale-free normalizer: prefer target variance, fall back to energy, then n.
    normalizer = if target_is_variable
        m2_y
    elseif target_energy > zero_l
        target_energy
    else
        n_l
    end

    raw_nmse = squared_error / normalizer
    if !isfinite(raw_nmse) || raw_nmse < zero_l
        return L(Inf)
    end
    fidelity_term = log1p(raw_nmse)

    # Bounded shape term: normalized residual after the best affine map a*p + b.
    shape_term = zero_l
    if target_is_variable
        if m2_p > prediction_floor
            correlation = covariance / sqrt(m2_p) / sqrt(m2_y)
            if !isfinite(correlation)
                return L(Inf)
            end
            bounded_correlation = min(abs(correlation), one_l)
            shape_term = one_l - bounded_correlation * bounded_correlation
        else
            # A constant prediction can match only the target mean: worst shape.
            shape_term = one_l
        end
    end

    # Continuation: structural guidance is strong far away, vanishes near a fit.
    max_shape_weight = L(0.5)
    shape_weight = max_shape_weight * raw_nmse / (one_l + raw_nmse)

    loss_value = (one_l - shape_weight) * fidelity_term + shape_weight * shape_term
    return isfinite(loss_value) ? L(max(loss_value, zero_l)) : L(Inf)
end
```
