## Parent survival operator 1
```julia
"""
    pareto_niche_crowding_survival_gen25_8(pop, options; exclude_indices=Int[])

Survival ("who dies") operator that generalizes age-regularized evolution into a
**Pareto-aware, niche-crowding** replacement rule.

Core idea
=========
Age-regularized evolution (the default) always kills the oldest member. This keeps
turnover high but is blind to *what* is being discarded: it happily deletes a
unique, non-dominated building block (e.g. the only simple, low-loss skeleton of the
ground-truth expression) while leaving behind dozens of redundant clones of the
current champion. Since ground-truth recovery usually proceeds by *growing* a
correct low-complexity skeleton, protecting the accuracy/complexity Pareto front and
deleting redundancy instead of age is expected to help exact-recovery rates.

This operator therefore assigns each member a *replacement priority* and removes the
member with the highest priority. Priority rewards:

  1. **Age** (`age_frac`): keeps the age-regularization pressure that prevents
     stagnation and lets bad-but-lucky members eventually leave.
  2. **Pareto dominatedness** (`dom_frac`): the number of members that are at least as
     simple *and* at least as accurate. Heavily dominated members contribute nothing
     to the front and are the cheapest to lose.
  3. **Niche crowding** (`crowd_frac`): occupancy of the member's exact-complexity
     bin. This spreads the population over the complexity axis, which is exactly the
     axis PySR must search to find the true expression size.
  4. **Redundancy / clones**: members sharing the same complexity *and* (numerically)
     the same loss are near-certain duplicates; all but one are free to delete.

and it *protects*:

  * members that are non-dominated **and** unique (the Pareto front is the search's memory),
  * the newest ~10% of members (a brand-new offspring gets at least one chance to be
    selected as a parent, otherwise mutation feedback is destroyed),
  * the current best-cost champion (near-absolute protection, i.e. elitism inside the
    population itself).

Steps
=====
1. Cache complexity, loss, birth and cost of every member (NaN losses/costs -> `Inf`).
2. Count Pareto dominators for each member (O(n^2), n is the population size, so this
   is negligible next to a single loss evaluation).
3. Detect clones (same complexity, relatively-equal loss) and complexity-niche counts.
4. Combine the normalized terms into a replacement priority with the protections above,
   add a tiny random jitter for tie-breaking (avoids systematically deleting the
   lowest-index member of a tied group), and return the argmax.
5. `exclude_indices` are never returned; if everything is excluded we fall back to the
   first admissible index (and to `1` in the degenerate case).

Assumptions / heuristics
========================
* Lower `loss` and lower complexity are both better; `cost` is the parsimony-adjusted
  objective already computed by SymbolicRegression.jl.
* Weights were chosen so that (a) a unique Pareto-front member essentially never dies,
  (b) among the remaining members, redundancy dominates age, and (c) age still breaks
  ties so the population cannot freeze.
* Uses `rand()` only for jitter, so behaviour is robust to the RNG stream.
"""
function pareto_niche_crowding_survival_gen25_8(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    n = pop.n
    n <= 1 && return 1

    # ---------------------------------------------------------------- caches
    complexities = Vector{Int}(undef, n)
    losses = Vector{Float64}(undef, n)
    costs = Vector{Float64}(undef, n)
    births = Vector{Float64}(undef, n)
    @inbounds for i in 1:n
        member = pop.members[i]
        complexities[i] = compute_complexity(member, options)
        l = Float64(member.loss)
        losses[i] = isnan(l) ? Inf : l          # treat invalid losses as worst possible
        c = Float64(member.cost)
        costs[i] = isnan(c) ? Inf : c
        births[i] = Float64(member.birth)
    end

    # --------------------------------------- Pareto dominance counts (2D front)
    # n_dominators[i] = #{j : j is <= as complex AND <= as lossy, strictly better once}
    n_dominators = zeros(Int, n)
    @inbounds for i in 1:n
        ci = complexities[i]
        li = losses[i]
        for j in 1:n
            j == i && continue
            cj = complexities[j]
            lj = losses[j]
            if cj <= ci && lj <= li && (cj < ci || lj < li)
                n_dominators[i] += 1
            end
        end
    end
    max_dom = maximum(n_dominators)

    # ------------------------------------------------- clone / duplicate detection
    # Same complexity + (relatively) same loss => almost surely the same formula.
    clone_count = zeros(Int, n)
    @inbounds for i in 1:(n - 1)
        for j in (i + 1):n
            if complexities[i] == complexities[j] && (
                losses[i] == losses[j] ||
                isapprox(losses[i], losses[j]; rtol=1e-9, atol=0.0)
            )
                clone_count[i] += 1
                clone_count[j] += 1
            end
        end
    end

    # ------------------------------------------- complexity-niche occupancy counts
    niche = Dict{Int,Int}()
    @inbounds for i in 1:n
        niche[complexities[i]] = get(niche, complexities[i], 0) + 1
    end
    max_niche = maximum(values(niche))

    # --------------------------------------------------------- age normalization
    bmin = minimum(births)
    bmax = maximum(births)
    brange = bmax - bmin

    # Youth shield: protect the newest ~10% (at least 1) so fresh offspring survive
    # long enough to be sampled as parents at least once.
    n_shield = max(1, div(n, 10))
    sorted_births = sort(births; rev=true)
    shield_birth = sorted_births[min(n_shield, n)]

    # Champion (best parsimony-adjusted cost) receives near-absolute protection.
    champion_idx = argmin_fast(costs)

    # ------------------------------------------------- replacement priority search
    best_idx = 0
    best_priority = -Inf
    @inbounds for i in 1:n
        (i in exclude_indices) && continue

        # older => closer to 1 (retains age-regularized turnover pressure)
        age_frac = brange > 0 ? (bmax - births[i]) / brange : 0.5
        # more dominated => closer to 1 (useless for the front)
        dom_frac = max_dom > 0 ? n_dominators[i] / max_dom : 0.0
        # more crowded complexity bin => closer to 1 (spread over the complexity axis)
        crowd_frac = max_niche > 1 ? (niche[complexities[i]] - 1) / (max_niche - 1) : 0.0

        priority = 1.0 * age_frac + 1.2 * dom_frac + 0.7 * crowd_frac

        # Pure redundancy: duplicates are the cheapest deletions available (saturating).
        priority += 1.5 * (min(clone_count[i], 3) / 3)

        # Protect unique, non-dominated members: they are the Pareto front / search memory.
        if n_dominators[i] == 0 && clone_count[i] == 0
            priority -= 4.0
        end

        # Do not immediately discard brand-new offspring.
        if births[i] >= shield_birth
            priority -= 2.5
        end

        # Population-level elitism for the current champion.
        if i == champion_idx
            priority -= 100.0
        end

        # Tiny jitter breaks exact ties without biasing toward low indices.
        priority += 0.05 * rand()

        if priority > best_priority
            best_priority = priority
            best_idx = i
        end
    end

    # ------------------------------------------------------------------ fallbacks
    if best_idx == 0
        # Everything was excluded (degenerate): return the first admissible index.
        @inbounds for i in 1:n
            (i in exclude_indices) || return i
        end
        return 1
    end

    return best_idx
end
```

## Parent survival operator 2
```julia
"""
    quality_aware_pareto_niche_survival_gen13_1_gen13_1(pop, options; exclude_indices=Int[])

Select the most expendable population member using age regularization, fitness,
redundancy, complexity-space crowding, lineage crowding, and Pareto protection.

The operator proceeds as follows:

1. Compute complexity, loss, cost, birth, and validity for every member.
2. Build tie-aware age and cost ranks, avoiding arbitrary rank differences among
   members with identical values.
3. Detect redundancy at two resolutions using `(complexity, rounded loss)`:
   a fine signature identifies near-certain clones, while a coarser signature
   identifies a broader functional niche. Only the best, newest representative
   of each signature is protected; unlike the parent, duplicate pressure is not
   applied equally to the representative that should be retained.
4. Measure both exact-complexity occupancy and local `±1` complexity-band
   density. Crowding pressure is quality-gated so poor members are removed from
   dense regions before useful building blocks.
5. Mildly penalize overrepresented positive parent IDs, while ignoring sentinel
   parent IDs commonly assigned to initial population members.
6. Construct a representative Pareto frontier in `(complexity, loss)`. Strictly
   non-dominated points are protected, but points providing only a numerically
   tiny loss improvement receive weaker protection.
7. Softly protect the best-cost and best-loss members. These protections decay
   with age, replacing the parent's effectively immortal champion with elites
   that are stable when young but can eventually turn over when stale and
   crowded.

Non-finite members receive a dominant deletion penalty and are never selected as
elites. A small random jitter prevents deterministic tie cycles. Excluded
indices are masked once and are never selected unless every population slot is
excluded, in which case index `1` is the unavoidable degenerate fallback.
"""
function quality_aware_pareto_niche_survival_gen13_1_gen13_1(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    n = pop.n
    n <= 1 && return 1

    # Mask exclusions once rather than repeatedly scanning exclude_indices.
    excluded = falses(n)
    for idx in exclude_indices
        if 1 <= idx <= n
            excluded[idx] = true
        end
    end
    all(excluded) && return 1

    # ------------------------------------------------------------------
    # 1) Gather per-member statistics.
    # ------------------------------------------------------------------
    comp = Vector{Int}(undef, n)
    lossf = Vector{Float64}(undef, n)
    costf = Vector{Float64}(undef, n)
    births = Vector{Float64}(undef, n)
    broken = falses(n)

    for i in 1:n
        m = pop.members[i]
        comp[i] = compute_complexity(m, options)

        raw_loss = Float64(m.loss)
        raw_cost = Float64(m.cost)
        broken[i] = !isfinite(raw_loss) || !isfinite(raw_cost)

        lossf[i] = isfinite(raw_loss) ? raw_loss : Inf
        costf[i] = isfinite(raw_cost) ? raw_cost : Inf
        births[i] = Float64(m.birth)
    end

    # ------------------------------------------------------------------
    # 2) Tie-aware ranks in [0, 1].
    #    Age: 0 is newest, 1 is oldest.
    #    Cost: 0 is best, 1 is worst.
    # ------------------------------------------------------------------
    denom = Float64(max(1, n - 1))
    age_rank = Vector{Float64}(undef, n)
    cost_rank = Vector{Float64}(undef, n)

    age_order = sortperm(births; rev=true)
    k = 1
    while k <= n
        j = k
        while j < n && births[age_order[j + 1]] == births[age_order[k]]
            j += 1
        end
        rank_value = ((k - 1) + (j - 1)) / (2.0 * denom)
        for pos in k:j
            age_rank[age_order[pos]] = rank_value
        end
        k = j + 1
    end

    cost_order = sortperm(costf)
    k = 1
    while k <= n
        j = k
        while j < n && costf[cost_order[j + 1]] == costf[cost_order[k]]
            j += 1
        end
        rank_value = ((k - 1) + (j - 1)) / (2.0 * denom)
        for pos in k:j
            cost_rank[cost_order[pos]] = rank_value
        end
        k = j + 1
    end

    # ------------------------------------------------------------------
    # 3) Two-resolution functional redundancy.
    # ------------------------------------------------------------------
    fine_sigs = Vector{Tuple{Int,Float64}}(undef, n)
    coarse_sigs = Vector{Tuple{Int,Float64}}(undef, n)
    fine_counts = Dict{Tuple{Int,Float64},Int}()
    coarse_counts = Dict{Tuple{Int,Float64},Int}()

    for i in 1:n
        fine_loss = isfinite(lossf[i]) ? round(lossf[i]; sigdigits=8) : Inf
        coarse_loss = isfinite(lossf[i]) ? round(lossf[i]; sigdigits=5) : Inf

        fine_sigs[i] = (comp[i], fine_loss)
        coarse_sigs[i] = (comp[i], coarse_loss)
        fine_counts[fine_sigs[i]] = get(fine_counts, fine_sigs[i], 0) + 1
        coarse_counts[coarse_sigs[i]] =
            get(coarse_counts, coarse_sigs[i], 0) + 1
    end

    # Keep the lowest-loss, then lowest-cost, then newest representative.
    fine_rep = Dict{Tuple{Int,Float64},Int}()
    coarse_rep = Dict{Tuple{Int,Float64},Int}()

    for i in 1:n
        fs = fine_sigs[i]
        old = get(fine_rep, fs, 0)
        if old == 0 ||
           lossf[i] < lossf[old] ||
           (lossf[i] == lossf[old] && costf[i] < costf[old]) ||
           (lossf[i] == lossf[old] && costf[i] == costf[old] &&
            births[i] > births[old])
            fine_rep[fs] = i
        end

        cs = coarse_sigs[i]
        old = get(coarse_rep, cs, 0)
        if old == 0 ||
           lossf[i] < lossf[old] ||
           (lossf[i] == lossf[old] && costf[i] < costf[old]) ||
           (lossf[i] == lossf[old] && costf[i] == costf[old] &&
            births[i] > births[old])
            coarse_rep[cs] = i
        end
    end

    # ------------------------------------------------------------------
    # 4) Exact and local complexity-space crowding.
    # ------------------------------------------------------------------
    comp_counts = Dict{Int,Int}()
    for c in comp
        comp_counts[c] = get(comp_counts, c, 0) + 1
    end

    local_band = Vector{Int}(undef, n)
    max_exact_count = 1
    max_band_count = 1
    for i in 1:n
        exact_count = comp_counts[comp[i]]
        local_count =
            get(comp_counts, comp[i] - 1, 0) +
            exact_count +
            get(comp_counts, comp[i] + 1, 0)

        local_band[i] = local_count
        max_exact_count = max(max_exact_count, exact_count)
        max_band_count = max(max_band_count, local_count)
    end

    # ------------------------------------------------------------------
    # 5) Immediate-family crowding. Nonpositive IDs are treated as sentinels.
    # ------------------------------------------------------------------
    family_counts = Dict{Int,Int}()
    for i in 1:n
        parent = pop.members[i].parent
        if parent > 0
            family_counts[parent] = get(family_counts, parent, 0) + 1
        end
    end
    family_scale = max(3.0, sqrt(Float64(n)))

    # ------------------------------------------------------------------
    # 6) One quality representative per complexity, followed by a Pareto sweep.
    # ------------------------------------------------------------------
    best_at_complexity = Dict{Int,Int}()
    for i in 1:n
        c = comp[i]
        old = get(best_at_complexity, c, 0)
        if old == 0 ||
           lossf[i] < lossf[old] ||
           (lossf[i] == lossf[old] && costf[i] < costf[old]) ||
           (lossf[i] == lossf[old] && costf[i] == costf[old] &&
            births[i] > births[old])
            best_at_complexity[c] = i
        end
    end

    frontier_order = collect(values(best_at_complexity))
    sort!(frontier_order; by=i -> comp[i])

    nondom = falses(n)
    frontier_strength = zeros(Float64, n)
    best_loss_so_far = Inf

    for idx in frontier_order
        current_loss = lossf[idx]
        if isfinite(current_loss) && current_loss < best_loss_so_far
            nondom[idx] = true

            if isinf(best_loss_so_far)
                frontier_strength[idx] = 1.0
            else
                improvement = best_loss_so_far - current_loss
                meaningful_tol = 1.0e-12 + 1.0e-6 * abs(best_loss_so_far)
                # Tiny numerical improvements remain protected, but less strongly.
                frontier_strength[idx] =
                    improvement > meaningful_tol ? 1.0 : 0.35
            end
            best_loss_so_far = current_loss
        end
    end

    # Select finite best-cost and best-loss representatives. Ties favor simpler
    # and newer members, preventing an arbitrary old duplicate from being elite.
    best_cost_idx = 0
    best_loss_idx = 0
    for i in 1:n
        broken[i] && continue

        if best_cost_idx == 0 ||
           costf[i] < costf[best_cost_idx] ||
           (costf[i] == costf[best_cost_idx] &&
            comp[i] < comp[best_cost_idx]) ||
           (costf[i] == costf[best_cost_idx] &&
            comp[i] == comp[best_cost_idx] &&
            births[i] > births[best_cost_idx])
            best_cost_idx = i
        end

        if best_loss_idx == 0 ||
           lossf[i] < lossf[best_loss_idx] ||
           (lossf[i] == lossf[best_loss_idx] &&
            comp[i] < comp[best_loss_idx]) ||
           (lossf[i] == lossf[best_loss_idx] &&
            comp[i] == comp[best_loss_idx] &&
            costf[i] < costf[best_loss_idx]) ||
           (lossf[i] == lossf[best_loss_idx] &&
            comp[i] == comp[best_loss_idx] &&
            costf[i] == costf[best_loss_idx] &&
            births[i] > births[best_loss_idx])
            best_loss_idx = i
        end
    end

    # ------------------------------------------------------------------
    # 7) Score allowed members; the largest score is most expendable.
    # ------------------------------------------------------------------
    best_score = -Inf
    best_choice = 0

    for i in 1:n
        excluded[i] && continue

        fine_count = fine_counts[fine_sigs[i]]
        coarse_count = coarse_counts[coarse_sigs[i]]

        # A duplicate pair already receives substantial pressure; larger clone
        # groups saturate to prevent redundancy from overwhelming every signal.
        fine_duplicate_penalty = 0.0
        if fine_count > 1 && fine_rep[fine_sigs[i]] != i
            fine_duplicate_penalty =
                0.70 + 0.30 * min(1.0, (fine_count - 2) / 3.0)
        end

        # Coarse niches represent similar fitted behavior and receive only mild
        # pressure because distinct structures can occasionally share a loss.
        coarse_niche_penalty = 0.0
        if coarse_count > 1 && coarse_rep[coarse_sigs[i]] != i
            coarse_niche_penalty =
                min(1.0, (coarse_count - 1) / 4.0)
        end

        exact_crowding =
            max_exact_count > 1 ?
            (comp_counts[comp[i]] - 1) / Float64(max_exact_count - 1) : 0.0

        band_crowding =
            max_band_count > 1 ?
            (local_band[i] - 1) / Float64(max_band_count - 1) : 0.0

        parent = pop.members[i].parent
        family_count = parent > 0 ? get(family_counts, parent, 1) : 1
        family_crowding =
            min(1.0, (family_count - 1) / family_scale)

        # Crowding acts most strongly on poor members, so rare high-quality
        # building blocks are not discarded merely because their band is dense.
        quality_gate = 0.35 + 0.65 * cost_rank[i]

        score =
            1.10 * age_rank[i] +
            0.65 * cost_rank[i] +
            1.35 * fine_duplicate_penalty +
            0.35 * coarse_niche_penalty +
            quality_gate *
            (0.45 * exact_crowding + 0.35 * band_crowding) +
            0.30 * quality_gate * family_crowding

        # Invalid members dominate ordinary deletion candidates.
        broken[i] && (score += 5.0)

        # Protections decay with age so no finite member is permanently frozen.
        if nondom[i]
            pareto_age_factor = 1.0 - 0.55 * age_rank[i]
            pareto_quality = 0.40 + 0.60 * frontier_strength[i]
            score -= 1.15 * pareto_quality * pareto_age_factor
        end

        if i == best_cost_idx
            score -= 1.35 * (1.0 - 0.75 * age_rank[i])
        end
        if i == best_loss_idx
            score -= 0.75 * (1.0 - 0.65 * age_rank[i])
        end

        # Small enough to preserve the score ordering except near ties.
        score += 0.01 * rand()

        if score > best_score
            best_score = score
            best_choice = i
        end
    end

    return best_choice == 0 ? 1 : best_choice
end
```
