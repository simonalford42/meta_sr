```julia
"""
    affine_continuation_loss_init_8(tree, dataset, options)

Evaluate an expression using an adaptive combination of raw normalized error and an
affine-calibrated structural error. A single numerically stable Welford pass computes
the raw residual sum of squares and the moments needed to estimate the best affine
mapping `a * prediction + b` analytically. The two errors are variance-normalized and
compressed with `log1p`.

When raw error is large, up to 75% of the loss comes from the affine-calibrated error,
rewarding expressions that have the correct functional shape before their constants
are tuned. This scaffold fades continuously as raw error approaches zero, so only the
actual target predictions attain zero loss. Nearly constant targets disable the
affine scaffold, and variance floors avoid unstable calibration of constant
predictions. No expression-complexity penalty is included.
"""
function affine_continuation_loss_init_8(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = dataset.n
    if n <= 0 || length(prediction) != n || length(dataset.y) != n
        return L(Inf)
    end

    zero_l = zero(L)
    one_l = one(L)

    # Online means and centered sums avoid the cancellation of raw power sums.
    mean_p = zero_l
    mean_y = zero_l
    mean_d = zero_l
    m2_p = zero_l
    m2_y = zero_l
    m2_d = zero_l
    cov_py = zero_l

    @inbounds for i in 1:n
        p = L(prediction[i])
        y = L(dataset.y[i])
        d = p - y

        if !(isfinite(p) && isfinite(y) && isfinite(d))
            return L(Inf)
        end

        k = L(i)

        delta_p = p - mean_p
        mean_p += delta_p / k

        delta_y = y - mean_y
        mean_y += delta_y / k

        # This covariance update uses the newly updated target mean.
        cov_py += delta_p * (y - mean_y)
        m2_p += delta_p * (p - mean_p)
        m2_y += delta_y * (y - mean_y)

        delta_d = d - mean_d
        mean_d += delta_d / k
        m2_d += delta_d * (d - mean_d)
    end

    if !(isfinite(mean_p) && isfinite(mean_y) && isfinite(mean_d) &&
         isfinite(m2_p) && isfinite(m2_y) && isfinite(m2_d) &&
         isfinite(cov_py))
        return L(Inf)
    end

    # Centered sums can become microscopically negative from roundoff.
    m2_p = max(m2_p, zero_l)
    m2_y = max(m2_y, zero_l)
    m2_d = max(m2_d, zero_l)

    n_l = L(n)
    raw_sse = m2_d + n_l * mean_d * mean_d
    if !isfinite(raw_sse)
        return L(Inf)
    elseif raw_sse == zero_l
        return zero_l
    end

    # Relative floors suppress unstable affine fits to numerically constant data.
    y_reference = max(one_l, abs(mean_y))
    p_reference = max(one_l, abs(mean_p))
    scale_floor = n_l * eps(L) * y_reference * y_reference
    prediction_floor = n_l * eps(L) * p_reference * p_reference

    if !(isfinite(scale_floor) && isfinite(prediction_floor))
        return L(Inf)
    end

    scale_sse = max(m2_y, scale_floor)
    target_is_variable = m2_y > scale_floor

    affine_sse = raw_sse
    if target_is_variable
        if m2_p > prediction_floor
            # For least-squares a*p+b, explained SSE is cov(p,y)^2 / var(p).
            normalized_covariance = cov_py / sqrt(m2_p)
            explained_sse = normalized_covariance * normalized_covariance
            if !isfinite(explained_sse)
                return L(Inf)
            end
            affine_sse = clamp(m2_y - explained_sse, zero_l, m2_y)
        else
            # A constant prediction can only fit the target intercept.
            affine_sse = m2_y
        end
    end

    raw_nmse = raw_sse / scale_sse
    affine_nmse = affine_sse / scale_sse
    if !(isfinite(raw_nmse) && isfinite(affine_nmse))
        return L(Inf)
    end

    raw_term = log1p(max(raw_nmse, zero_l))
    affine_term = log1p(max(affine_nmse, zero_l))

    # Structural guidance is strong far from the target and vanishes near a fit.
    scaffold_strength = L(3) / L(4)
    blend = scaffold_strength * raw_nmse / (one_l + raw_nmse)
    loss_value = (one_l - blend) * raw_term + blend * affine_term

    return isfinite(loss_value) ? L(max(loss_value, zero_l)) : L(Inf)
end
```

## Parent loss operator 2
```julia
"""
    shape_guided_log_nmse_loss_gen3_3(tree, dataset, options)

Evaluate an expression with a scale-normalized MSE augmented by a lightweight
shape-matching signal. The direct error is normalized by target energy and
log-compressed so numerically poor candidates do not overwhelm useful structural
differences. A secondary `1 - r²` term measures agreement after the best implicit
affine rescaling, helping promising nonlinear structure survive before its
constants and offset are fully optimized.

Compared with the parent MSE, this loss is scale-independent, less dominated by
large aggregate errors, allocation-free, and provides affine-invariant shape
guidance. Online moments compute all statistics in one pass. Constant targets
disable the shape term, exact predictions return zero, and no expression
complexity penalty is included.
"""
function shape_guided_log_nmse_loss_gen3_3(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = length(dataset.y)
    if n == 0 || length(prediction) != n
        return L(Inf)
    end

    # Welford accumulators provide stable variances and covariance in one pass.
    mean_y = zero(L)
    mean_p = zero(L)
    m2_y = zero(L)
    m2_p = zero(L)
    covariance = zero(L)
    target_energy = zero(L)
    squared_error = zero(L)

    @inbounds for i in 1:n
        y = L(dataset.y[i])
        p = L(prediction[i])
        if !isfinite(y) || !isfinite(p)
            return L(Inf)
        end

        residual = p - y
        squared_error = muladd(residual, residual, squared_error)
        target_energy = muladd(y, y, target_energy)

        count = L(i)
        delta_y = y - mean_y
        delta_p = p - mean_p
        mean_y += delta_y / count
        mean_p += delta_p / count

        m2_y = muladd(delta_y, y - mean_y, m2_y)
        m2_p = muladd(delta_p, p - mean_p, m2_p)
        covariance = muladd(delta_p, y - mean_y, covariance)
    end

    # Preserve an exact zero optimum even if unrelated moment sums overflow.
    if squared_error == zero(L)
        return zero(L)
    end

    if !isfinite(squared_error) || !isfinite(target_energy) ||
       !isfinite(m2_y) || !isfinite(m2_p) || !isfinite(covariance)
        return L(Inf)
    end

    # Normalization preserves relative accuracy across differently scaled tasks.
    normalizer = target_energy > zero(L) ? target_energy : L(n)
    normalized_mse = squared_error / normalizer
    if !isfinite(normalized_mse)
        return L(Inf)
    end

    # This equals the normalized residual of the best affine fit a*p + b.
    shape_loss = zero(L)
    if m2_y > zero(L)
        shape_loss = one(L)
        if m2_p > zero(L)
            correlation = covariance / sqrt(m2_p) / sqrt(m2_y)
            if !isfinite(correlation)
                return L(Inf)
            end
            bounded_correlation = min(abs(correlation), one(L))
            shape_loss = one(L) - bounded_correlation * bounded_correlation
        end
    end

    # Direct fidelity remains dominant; the shape term supplies early guidance.
    loss_value = L(0.85) * log1p(normalized_mse) + L(0.15) * shape_loss
    return isfinite(loss_value) ? L(loss_value) : L(Inf)
end
```
