```julia
"""
    pareto_crowded_age_survival_init_4(
        pop::Population{T,L,N},
        options::AbstractOptions;
        exclude_indices::Vector{Int}=Int[],
    )::Int where {T,L,N}

Multi-objective survival operator for symbolic regression that balances age-regularized
evolutionary turnover, Pareto frontier preservation, and complexity-niche crowding penalty.

# Core Idea
Standard regularized evolution evicts the oldest individual regardless of whether it is a
breakthrough Pareto-optimal solution or a stagnant, heavily dominated duplicate.
This operator computes an eviction priority score for each candidate member by synthesizing
three evolutionary dynamics:
1. **Age Regularization (Exploration turnover)**: Older individuals receive higher eviction
   priority to ensure continuous turnover of genetic material.
2. **Pareto Dominance Shielding (Stepping-stone preservation)**: Solutions on or near the
   empirical Pareto front (Loss vs. Complexity) receive strong protection against premature
   eviction, preserving non-dominated stepping stones across the search space.
3. **Complexity-Niche Crowding & De-duplication**: Individuals that are dominated within
   heavily crowded complexity bins (multiple individuals at the identical complexity) are
   penalized and prioritized for eviction, freeing capacity for novel complexity niches.
4. **Immediate Sanitation**: Individuals with invalid (NaN/Inf) loss or cost are evicted
   immediately before any valid population member.

When all individuals in the population are non-dominated and uniquely distributed across
complexity bins, the dominance and crowding penalties become zero, and the operator smoothly
reduces to classic Age-Regularized Evolution.

# Returns
- `Int`: The 1-based index of the population member selected for replacement.
"""
function pareto_crowded_age_survival_init_4(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T,L,N}
    n = pop.n
    n <= 0 && return 1

    # Find first eligible index as a safe fallback for edge cases
    fallback_idx = 1
    for i in 1:n
        if !(i in exclude_indices)
            fallback_idx = i
            break
        end
    end

    # Pre-extract properties for fast vector processing
    complexities = Vector{Int}(undef, n)
    losses = Vector{Float64}(undef, n)
    births = Vector{Int}(undef, n)
    is_invalid = Vector{Bool}(undef, n)

    min_birth = typemax(Int)
    max_birth = typemin(Int)

    for i in 1:n
        member = pop.members[i]
        comp = compute_complexity(member, options)
        loss = Float64(member.loss)
        cost = Float64(member.cost)
        birth = member.birth

        complexities[i] = comp
        losses[i] = loss
        births[i] = birth

        # Check for non-finite / invalid numerical values
        invalid = isnan(loss) || isinf(loss) || isnan(cost) || isinf(cost)
        is_invalid[i] = invalid

        if !(i in exclude_indices)
            # Immediate eviction for non-finite solutions to sanitize the population
            invalid && return i

            if birth < min_birth
                min_birth = birth
            end
            if birth > max_birth
                max_birth = birth
            end
        end
    end

    # If all valid members are excluded or population is degenerate
    if min_birth > max_birth
        return fallback_idx
    end

    birth_span = max(1, max_birth - min_birth)

    # Compute Pareto Dominance count: how many valid members strictly dominate member i
    # Member j dominates member i if:
    #   (comp_j <= comp_i && loss_j <= loss_i) && (comp_j < comp_i || loss_j < loss_i)
    dom_counts = zeros(Int, n)
    max_dom = 0
    for i in 1:n
        (i in exclude_indices || is_invalid[i]) && continue
        comp_i = complexities[i]
        loss_i = losses[i]
        d_count = 0
        for j in 1:n
            (j == i || is_invalid[j]) && continue
            comp_j = complexities[j]
            loss_j = losses[j]
            if (comp_j <= comp_i && loss_j <= loss_i) && (comp_j < comp_i || loss_j < loss_i)
                d_count += 1
            end
        end
        dom_counts[i] = d_count
        if d_count > max_dom
            max_dom = d_count
        end
    end

    # Compute Complexity-Niche Crowding & Intra-bin Loss Rank:
    # Measures how crowded member i's complexity bin is and how sub-optimal member i is within it
    crowding_scores = zeros(Float64, n)
    for i in 1:n
        (i in exclude_indices || is_invalid[i]) && continue
        comp_i = complexities[i]
        loss_i = losses[i]

        bin_count = 0
        worse_in_bin_count = 0

        for j in 1:n
            is_invalid[j] && continue
            if complexities[j] == comp_i
                bin_count += 1
                if losses[j] < loss_i
                    worse_in_bin_count += 1
                end
            end
        end

        # Crowding penalty is non-zero only if multiple individuals share the same complexity
        if bin_count > 1
            relative_rank = worse_in_bin_count / (bin_count - 1)
            density_factor = bin_count / n
            crowding_scores[i] = relative_rank * density_factor
        end
    end

    # Compute composite eviction score:
    # Higher score -> higher priority for replacement
    best_eviction_score = -Inf
    best_idx = fallback_idx

    # Weight parameters for multi-objective eviction
    w_age = 0.40      # Base aging pressure for exploration turnover
    w_dom = 0.35      # Pareto dominance penalty (dominated solutions get evicted first)
    w_crowd = 0.25    # Complexity niche crowding penalty (prunes redundant bin duplicates)

    for i in 1:n
        (i in exclude_indices) && continue

        # Normalized age: 1.0 for oldest, 0.0 for youngest
        norm_age = (max_birth - births[i]) / birth_span

        # Normalized dominance: 0.0 for Pareto-optimal non-dominated solutions
        norm_dom = max_dom > 0 ? (dom_counts[i] / max_dom) : 0.0

        # Crowding score: 0.0 for bin champions or unique complexity bins
        norm_crowd = crowding_scores[i]

        eviction_score = w_age * norm_age + w_dom * norm_dom + w_crowd * norm_crowd

        if eviction_score > best_eviction_score
            best_eviction_score = eviction_score
            best_idx = i
        end
    end

    return best_idx
end
```

## Parent survival operator 2
```julia
"""
    pareto_age_clone_niche_survival_gen1_7(
        pop::Population{T,L,N},
        options::AbstractOptions;
        exclude_indices::Vector{Int}=Int[],
    )::Int where {T,L,N}

Survival (replacement) operator synthesizing age-regularized turnover, Pareto-front
shielding, complexity-niche crowding, and explicit clone detection.

# Core Idea
Both parent operators recognize that blindly killing the oldest member destroys
stepping-stone solutions, while purely greedy replacement collapses diversity. This
operator merges their strongest signals into one expendability score, while retaining
each parent's safety nets:

1. **Immediate Sanitation** (from parent 1): any *non-excluded* member with
   non-finite loss/cost is evicted immediately, before anything else is considered.
2. **Elitism** (from parent 2): the single best-cost member is (almost) never evicted.
3. **Pareto Dominance Pressure** (from both): members dominated by many others in
   (complexity, loss) space accumulate eviction priority; non-dominated members are
   shielded.
4. **Exact-Duplicate ("Clone") Penalty** (from parent 2): among members sharing
   identical complexity and (numerically) identical loss, only the earliest-born
   representative is left unpenalized — the rest are pure redundancy and are
   prioritized for removal, harder than mere Pareto domination.
5. **Continuous Complexity-Niche Crowding** (from parent 1): within a complexity bin,
   a member's relative loss rank times the bin's population share yields a smooth
   crowding penalty, concentrating eviction pressure on over-populated, sub-optimal
   niches without requiring exact ties.
6. **Age Regularization with Stochastic Fallback** (from both): age contributes a
   continuous eviction bonus, and with small probability the operator instead performs
   a pure oldest-first eviction among unprotected, non-elite members — guaranteeing
   that no individual can survive forever purely by sitting in a sparse niche.

The composite score is a weighted sum of normalized age, dominance, crowding, and
clone-penalty terms, with large discounts for Pareto-front members and the elite.

# Heuristics / Assumptions
* `(complexity, loss)` (not `cost`) defines dominance/clone niches, decoupling this
  from the fluctuating adaptive parsimony term.
* Complexity bins are exact integer complexities.
* Clone detection uses a relative tolerance (`1e-10`) on loss equality.
* `exclude_indices` are always respected and never returned.

# Returns
- `Int`: the 1-based index of the population member selected for replacement.
"""
function pareto_age_clone_niche_survival_gen1_7(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T,L,N}
    n = pop.n
    n <= 0 && return 1

    # Safe fallback: first eligible (non-excluded) index.
    fallback_idx = 1
    for i in 1:n
        if !(i in exclude_indices)
            fallback_idx = i
            break
        end
    end
    n == 1 && return fallback_idx

    # ------------------------------------------------------------------
    # 1. Gather per-member statistics; flag invalid members for
    #    immediate sanitation.
    # ------------------------------------------------------------------
    complexities = Vector{Int}(undef, n)
    losses = Vector{Float64}(undef, n)
    costs = Vector{Float64}(undef, n)
    births = Vector{Float64}(undef, n)
    is_invalid = Vector{Bool}(undef, n)

    min_birth = Inf
    max_birth = -Inf

    for i in 1:n
        m = pop.members[i]
        complexities[i] = compute_complexity(m, options)
        l = Float64(m.loss)
        c = Float64(m.cost)
        births[i] = Float64(m.birth)
        is_invalid[i] = isnan(l) || isinf(l) || isnan(c) || isinf(c)
        losses[i] = is_invalid[i] ? Inf : l
        costs[i] = is_invalid[i] ? Inf : c

        if !(i in exclude_indices)
            # Immediate eviction of the first non-excluded broken member.
            is_invalid[i] && return i
            if births[i] < min_birth
                min_birth = births[i]
            end
            if births[i] > max_birth
                max_birth = births[i]
            end
        end
    end

    if min_birth > max_birth
        # All valid members were excluded; nothing sensible left to score.
        return fallback_idx
    end
    birth_range = max(max_birth - min_birth, 1.0)

    # Elite (best-cost, valid) member: protected via a huge score discount.
    best_cost_idx = fallback_idx
    best_cost_val = Inf
    for i in 1:n
        (i in exclude_indices || is_invalid[i]) && continue
        if costs[i] < best_cost_val
            best_cost_val = costs[i]
            best_cost_idx = i
        end
    end

    # ------------------------------------------------------------------
    # 2. Pairwise pass over valid members: dominance counts, clone
    #    counts, and complexity-bin statistics for crowding.
    # ------------------------------------------------------------------
    n_dom = zeros(Int, n)
    n_clone = zeros(Int, n)
    bin_count = zeros(Int, n)          # members sharing i's complexity
    worse_in_bin = zeros(Int, n)       # members in bin with strictly worse (higher) loss

    max_dom = 0
    for i in 1:n
        is_invalid[i] && continue
        ci = complexities[i]
        li = losses[i]
        bi = births[i]
        for j in 1:n
            (j == i || is_invalid[j]) && continue
            cj = complexities[j]
            lj = losses[j]

            # Strict Pareto dominance: j dominates i.
            if (cj <= ci && lj <= li) && (cj < ci || lj < li)
                n_dom[i] += 1
            end

            if cj == ci
                bin_count[i] += 1
                if lj < li
                    worse_in_bin[i] += 1  # counts members better than i for crowding rank
                end
                # Exact structural clone: identical complexity & (numerically) loss.
                same_loss = abs(lj - li) <= 1e-10 * (1.0 + abs(li))
                if same_loss && (bj_is_earlier_ac(births[j], bi, j, i))
                    n_clone[i] += 1
                end
            end
        end
        if n_dom[i] > max_dom
            max_dom = n_dom[i]
        end
        bin_count[i] += 1  # include self in bin population count
    end

    # A member sits on the Pareto front (and is not a redundant clone) if nothing
    # strictly dominates it and it is not a later duplicate of an equal twin.
    protected = [(!is_invalid[i] && n_dom[i] == 0 && n_clone[i] == 0) for i in 1:n]

    # ------------------------------------------------------------------
    # 3. Stochastic pure-age fallback: occasionally force oldest-first
    #    eviction among unprotected, non-elite members to guarantee
    #    turnover even in sparse/well-spread populations.
    # ------------------------------------------------------------------
    p_age = 0.15
    if rand() < p_age
        best_idx = 0
        best_birth = Inf
        for i in 1:n
            (i in exclude_indices) && continue
            i == best_cost_idx && continue
            protected[i] && continue
            if births[i] < best_birth
                best_birth = births[i]
                best_idx = i
            end
        end
        best_idx != 0 && return best_idx
        # else fall through to full scoring (protections relaxed there via discounts)
    end

    # ------------------------------------------------------------------
    # 4. Composite expendability score.
    # ------------------------------------------------------------------
    w_age = 0.30      # aging pressure for turnover
    w_dom = 0.30      # Pareto dominance penalty
    w_crowd = 0.20    # continuous complexity-niche crowding penalty
    w_clone = 0.20     # exact-duplicate penalty (applied per-clone, on top of dominance)

    best_score = -Inf
    best_idx = fallback_idx

    for i in 1:n
        (i in exclude_indices) && continue

        if is_invalid[i]
            # Should already have been caught, but keep as a safety net.
            return i
        end

        norm_age = (births[i] - min_birth >= 0) ? (max_birth - births[i]) / birth_range : 0.0
        norm_dom = max_dom > 0 ? (n_dom[i] / max_dom) : 0.0

        crowd = 0.0
        if bin_count[i] > 1
            relative_rank = worse_in_bin[i] / (bin_count[i] - 1)
            density_factor = bin_count[i] / n
            crowd = relative_rank * density_factor
        end

        clone_penalty = n_clone[i] > 0 ? 1.0 : 0.0  # binary: any clone-of-earlier flag

        score = w_age * norm_age + w_dom * norm_dom + w_crowd * crowd + w_clone * clone_penalty

        # Shield genuine Pareto-front, non-redundant members.
        if protected[i]
            score -= 2.0
        end
        # Elitism: effectively never replace the best-cost member.
        if i == best_cost_idx
            score -= 1.0e6
        end

        # Small tie-break jitter.
        score += 1e-6 * rand()

        if score > best_score
            best_score = score
            best_idx = i
        end
    end

    return best_idx
end

"""
    bj_is_earlier_ac(bj, bi, j, i)

Tie-break helper for clone detection: returns `true` when member `j` should be treated
as the earlier (surviving) representative of a clone cluster, using birth order first
and index as a deterministic fallback for simultaneous births.
"""
function bj_is_earlier_ac(bj::Float64, bi::Float64, j::Int, i::Int)
    return bj < bi || (bj == bi && j < i)
end
```
