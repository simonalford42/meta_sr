```julia
"""
Select a parent using a lineage-guarded Pareto-breakthrough tournament.

The operator samples a standard tournament, evaluates each member by raw loss,
complexity, penalized cost, and complexity-niche frequency, then counts how many
competitors dominate it in loss-complexity space. It rewards candidates that
achieve a substantial loss improvement over every simpler sampled expression,
which targets promising "breakthroughs" on the evolving Pareto frontier.
Underrepresented complexity niches, recent individuals, and sparsely represented
lineages receive modest protection to preserve stepping stones. Selection remains
stochastic through positive weights for every candidate, while Pareto dominance
and fitness ranks keep strongly inferior members unlikely to reproduce. The
heuristic assumes lower loss and cost are better and that useful ground-truth
subexpressions often appear as unusually effective improvements per added
complexity.
"""
function lineage_guarded_pareto_breakthrough_selection_gen2_4(
    pop::Population{T,L,N},
    running_search_statistics::RunningSearchStatistics,
    options::AbstractOptions,
)::PopMember{T,L,N} where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    tournament_size = min(pop.n, max(2, options.tournament_selection_n))
    sample = StatsBase.sample(pop.members, tournament_size; replace=false)
    n = length(sample)

    n == 1 && return sample[1]

    losses = Vector{Float64}(undef, n)
    costs = Vector{Float64}(undef, n)
    complexities = Vector{Int}(undef, n)
    frequencies = Vector{Float64}(undef, n)

    for i in 1:n
        member = sample[i]

        # Sanitize objectives so failed evaluations are always unattractive.
        loss = Float64(member.loss)
        cost = Float64(member.cost)
        losses[i] = isfinite(loss) ? loss : Inf
        costs[i] = isfinite(cost) ? cost : Inf

        size = compute_complexity(member, options)
        complexities[i] = size

        frequency = if 0 < size <= options.maxsize &&
                       size <= length(running_search_statistics.normalized_frequencies)
            Float64(running_search_statistics.normalized_frequencies[size])
        else
            0.0
        end
        frequencies[i] = isfinite(frequency) ? max(0.0, frequency) : 0.0
    end

    loss_ranks = zeros(Int, n)
    cost_ranks = zeros(Int, n)
    domination_counts = zeros(Int, n)
    niche_counts = ones(Int, n)
    lineage_counts = ones(Int, n)

    # Rank objectives, estimate local niche density, and count Pareto dominators.
    for i in 1:n
        for j in 1:n
            i == j && continue

            loss_ranks[i] += losses[j] < losses[i]
            cost_ranks[i] += costs[j] < costs[i]
            niche_counts[i] += complexities[j] == complexities[i]

            no_worse = losses[j] <= losses[i] &&
                       complexities[j] <= complexities[i]
            strictly_better = losses[j] < losses[i] ||
                              complexities[j] < complexities[i]
            domination_counts[i] += no_worse && strictly_better

            # Penalize crowded sibling groups and direct parent-child clusters.
            same_parent = sample[i].parent > 0 &&
                          sample[i].parent == sample[j].parent
            direct_relative = (sample[i].ref > 0 &&
                               sample[j].parent == sample[i].ref) ||
                              (sample[j].ref > 0 &&
                               sample[i].parent == sample[j].ref)
            lineage_counts[i] += same_parent || direct_relative
        end
    end

    # Measure whether each member is a genuine accuracy breakthrough over all
    # strictly simpler alternatives in this tournament.
    breakthroughs = zeros(Float64, n)
    for i in 1:n
        best_simpler_loss = Inf
        for j in 1:n
            if complexities[j] < complexities[i] && losses[j] < best_simpler_loss
                best_simpler_loss = losses[j]
            end
        end

        if isfinite(best_simpler_loss) && isfinite(losses[i]) &&
           losses[i] < best_simpler_loss
            scale = max(
                abs(best_simpler_loss) + abs(losses[i]),
                eps(Float64),
            )
            breakthroughs[i] = clamp(
                (best_simpler_loss - losses[i]) / scale,
                0.0,
                1.0,
            )
        end
    end

    min_birth = minimum(member.birth for member in sample)
    max_birth = maximum(member.birth for member in sample)
    birth_span = max_birth - min_birth

    p = clamp(Float64(options.tournament_selection_p), 0.0, 1.0)
    pressure = 0.75 + 2.25 * p
    rank_denominator = Float64(max(1, n - 1))
    dominance_scale = max(1.0, sqrt(Float64(n)))
    parsimony_scale = max(0.0, Float64(options.adaptive_parsimony_scaling))

    weights = Vector{Float64}(undef, n)
    for i in 1:n
        loss_rank = Float64(loss_ranks[i]) / rank_denominator
        cost_rank = Float64(cost_ranks[i]) / rank_denominator

        # A mixture of raw-loss and regularized-cost quality avoids committing
        # exclusively to either highly complex fits or overly simple formulas.
        quality_factor = 0.20 +
                         0.45 * exp(-pressure * loss_rank) +
                         0.35 * exp(-pressure * cost_rank)

        pareto_factor = exp(
            -pressure * Float64(domination_counts[i]) / dominance_scale,
        )
        breakthrough_factor = 1.0 + 3.0 * breakthroughs[i]

        # Protect sparse complexity niches both locally and, when requested,
        # according to the search-wide adaptive frequency statistics.
        local_niche_factor = inv(sqrt(Float64(niche_counts[i])))
        global_niche_factor = if options.use_frequency_in_tournament
            inv(sqrt(1.0 + parsimony_scale * frequencies[i]))
        else
            1.0
        end

        lineage_factor = inv(sqrt(Float64(lineage_counts[i])))

        # Briefly protect fresh mutations so potentially useful stepping stones
        # can reproduce before being erased by mature lineages.
        recency = birth_span > 0 ?
            Float64(sample[i].birth - min_birth) / Float64(birth_span) : 0.5
        youth_factor = 0.8 + 0.4 * recency

        weights[i] = quality_factor *
                     pareto_factor *
                     breakthrough_factor *
                     local_niche_factor *
                     global_niche_factor *
                     lineage_factor *
                     youth_factor
    end

    total_weight = sum(weights)
    if !isfinite(total_weight) || total_weight <= 0.0
        return sample[argmin_fast(costs)]
    end

    chosen_idx = StatsBase.sample(StatsBase.Weights(weights, total_weight))
    return sample[chosen_idx]
end
```

## Parent selection operator 2
```julia
# Custom Selection: tournament_selection
# =======================================
# Default selection strategy: tournament selection with adaptive parsimony,
# mirroring `default_selection` in CustomSelection.jl (which is itself a
# self-contained reimplementation of `best_of_sample` from Population.jl).
# Exposed here as a named custom operator so the meta-evolution loop has a
# concrete parent to refine from.

function tournament_selection(
    pop::Population{T,L,N},
    running_search_statistics::RunningSearchStatistics,
    options::AbstractOptions,
)::PopMember{T,L,N} where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    sample = StatsBase.sample(pop.members, options.tournament_selection_n; replace=false)
    n = length(sample)
    p = options.tournament_selection_p

    adjusted_costs = Vector{L}(undef, n)
    if options.use_frequency_in_tournament
        adaptive_parsimony_scaling = L(options.adaptive_parsimony_scaling)
        for i in 1:n
            member = sample[i]
            size = compute_complexity(member, options)
            frequency = if (0 < size <= options.maxsize)
                L(running_search_statistics.normalized_frequencies[size])
            else
                L(0)
            end
            adjusted_costs[i] = member.cost * exp(adaptive_parsimony_scaling * frequency)
        end
    else
        for i in 1:n
            adjusted_costs[i] = sample[i].cost
        end
    end

    chosen_idx = if p == 1.0
        argmin_fast(adjusted_costs)
    else
        k = collect(0:(n - 1))
        prob_each = p * ((1 - p) .^ k)
        weights = StatsBase.Weights(prob_each, sum(prob_each))
        tournament_winner = StatsBase.sample(weights)
        if tournament_winner == 1
            argmin_fast(adjusted_costs)
        else
            bottomk_fast(adjusted_costs, tournament_winner)[2][end]
        end
    end
    return copy(sample[chosen_idx])
end
```
