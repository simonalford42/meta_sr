```julia
"""
    residual_guided_basis_augmentation_gen9_8(tree, dataset, options, nfeatures, rng)

A data-aware "boosting" mutation: it looks at what the current expression gets *wrong*
and appends the single basis function that best explains the leftover error, with its
coefficient fitted by least squares on the training data.

Steps:
1. Evaluate the tree on (a strided subsample of) `dataset.X`. If evaluation fails or
   produces non-finite values, switch to a **repair** branch: locate the *smallest*
   subtree that itself evaluates to NaN/Inf on the data and replace it with a benign
   constant leaf (`avg_y` or `1`). This removes the numerically-broken part of the tree
   while keeping the rest of the (potentially useful) structure.
2. Otherwise compute the residual `r = y - ŷ`. If the residual is already negligible
   compared to `std(y)`, there is nothing to gain and the tree is returned unchanged.
3. Build a small library of candidate basis functions and score each by `|cor(basis, r)|`:
   (a) every raw feature `x_f`; (b) each unary operator applied to the few
   most residual-correlated features (`g(x_f)`); (c) products `x_f * x_g` among those
   same top features (including squares). Restricting (b)/(c) to the top features keeps
   the cost O(nfeatures · n) even with hundreds of features.
4. Pick a candidate *stochastically* from the top-3 scorers (probability ∝ score) so the
   operator stays exploratory rather than greedy/deterministic, then fit
   `r ≈ c·basis + b` by (centered) least squares. `c` is snapped to a nearby integer when
   it is close, which helps recover exact ground-truth coefficients.
5. Return `tree + c·basis` (plus the intercept `b` if it is non-negligible and the size
   budget allows). Requires `+` in the operator set; `*` is used for the coefficient when
   available, otherwise the basis is added unscaled.

Assumptions/heuristics: correlation with the residual is a cheap proxy for "which term is
missing"; a least-squares coefficient on a subsample generalizes; the extra complexity is
justified only if the parent search accepts it (the normal selection pressure decides).
Sample weights are ignored for simplicity.
"""
function residual_guided_basis_augmentation_gen9_8(
    tree::N,
    dataset,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # ---- basic guards -------------------------------------------------------
    dataset.y === nothing && return tree
    n = dataset.n
    n < 5 && return tree
    nfeatures < 1 && return tree

    # Subsample columns (samples) for speed; keep a strided view of the data.
    max_pts = 512
    if n <= max_pts
        Xs = dataset.X
        ys = dataset.y
    else
        idx = round.(Int, range(1, n; length=max_pts))
        Xs = dataset.X[:, idx]
        ys = dataset.y[idx]
    end
    ns = length(ys)

    # ---- 1. evaluate current tree ------------------------------------------
    pred, ok = eval_tree_array(tree, Xs, options.operators)

    if !ok || any(!isfinite, pred)
        # ---- REPAIR BRANCH: excise the smallest NaN/Inf-producing subtree ----
        worst = Ref{Any}(nothing)
        worst_size = Ref(typemax(Int))
        any(tree) do t
            if t.degree > 0
                sub, sok = eval_tree_array(t, Xs, options.operators)
                if !sok || any(!isfinite, sub)
                    sz = count_nodes(t)
                    if sz < worst_size[]
                        worst_size[] = sz
                        worst[] = t
                    end
                end
            end
            return false  # never short-circuit: visit every node
        end
        bad = worst[]
        bad === nothing && return tree
        # Replace the offending subtree with a harmless constant (mean of y, or 1).
        safe_val = dataset.avg_y === nothing ? one(T) : T(dataset.avg_y)
        repl = constructorof(N)(T; val=safe_val)
        if bad === tree
            return repl  # the whole tree was broken -> new root
        else
            set_node!(bad, repl)
            return tree
        end
    end

    # ---- 2. residual --------------------------------------------------------
    y64 = Float64.(ys)
    r64 = y64 .- Float64.(pred)
    sy = std(y64)
    sr = std(r64)
    (isfinite(sr) && sr > 1e-10 * max(1.0, abs(mean(y64)))) || return tree
    # Already essentially perfect on this subsample: nothing to add.
    (sy > 0 && sr < 1e-6 * sy) && return tree

    # Operator indices we need.
    plus_idx = findfirst(op -> op == (+), options.operators.binops)
    plus_idx === nothing && return tree            # cannot append a term without `+`
    mul_idx = findfirst(op -> op == (*), options.operators.binops)
    n_una = options.nops[1]

    # Size budget (do not blow past maxsize).
    maxsize = hasproperty(options, :maxsize) ? options.maxsize : 30
    base_size = count_nodes(tree)
    base_size + 4 > maxsize && return tree

    # ---- 3. candidate library, scored by |cor(basis, residual)| -------------
    # kinds: 1 = raw feature, 2 = unary(feature), 3 = feature * feature
    kinds = Int[]
    f1s = Int[]
    f2s = Int[]
    uops = Int[]
    vals = Vector{Vector{Float64}}()
    scores = Float64[]

    function push_candidate!(kind, f1, f2, uop, v)
        # Reject non-finite or constant candidate columns; else score by |cor|.
        all(isfinite, v) || return nothing
        sv = std(v)
        (isfinite(sv) && sv > 1e-12) || return nothing
        c = cor(v, r64)
        isfinite(c) || return nothing
        push!(kinds, kind); push!(f1s, f1); push!(f2s, f2); push!(uops, uop)
        push!(vals, v); push!(scores, abs(c))
        return nothing
    end

    # (a) raw features
    for f in 1:nfeatures
        push_candidate!(1, f, 0, 0, Float64.(@view Xs[f, :]))
    end
    isempty(scores) && return tree

    # Top few features (by |cor| with residual) get nonlinear expansion.
    raw_order = sortperm(scores; rev=true)
    ntop = min(4, length(raw_order))
    topf = [f1s[i] for i in raw_order[1:ntop]]

    # (b) unary transforms of the top features
    for f in topf, k in 1:n_una
        g = options.operators.unaops[k]
        v = try
            Float64.(g.(@view Xs[f, :]))
        catch
            continue
        end
        push_candidate!(2, f, 0, k, v)
    end

    # (c) pairwise products (and squares) among the top features
    if mul_idx !== nothing
        for i in 1:ntop, j in i:ntop
            fa, fb = topf[i], topf[j]
            v = Float64.(@view Xs[fa, :]) .* Float64.(@view Xs[fb, :])
            push_candidate!(3, fa, fb, 0, v)
        end
    end

    # ---- 4. stochastic pick among the top-3 candidates ---------------------
    order = sortperm(scores; rev=true)
    K = min(3, length(order))
    top = order[1:K]
    wsum = sum(scores[i] for i in top)
    pick = top[1]
    if wsum > 0
        u = rand(rng) * wsum
        acc = 0.0
        for i in top
            acc += scores[i]
            if u <= acc
                pick = i
                break
            end
        end
    end
    v = vals[pick]

    # Least-squares fit of r ≈ c * v + b (centered, numerically safe).
    mv = mean(v)
    mr = mean(r64)
    vc = v .- mv
    denom = sum(abs2, vc)
    denom > 1e-14 || return tree
    c = sum(vc .* (r64 .- mr)) / denom
    b = mr - c * mv
    (isfinite(c) && isfinite(b)) || return tree
    abs(c) > 1e-12 || return tree
    # Snap near-integer coefficients (helps hit exact ground-truth constants).
    if abs(c - round(c)) < 0.02 * max(1.0, abs(c))
        c = round(c)
    end

    # ---- 5. assemble  tree + c*basis (+ b) ---------------------------------
    kind = kinds[pick]
    basis = if kind == 1
        constructorof(N)(T; feature=f1s[pick])
    elseif kind == 2
        constructorof(N)(;
            op=uops[pick], children=(constructorof(N)(T; feature=f1s[pick]),)
        )
    else
        constructorof(N)(;
            op=mul_idx,
            children=(
                constructorof(N)(T; feature=f1s[pick]),
                constructorof(N)(T; feature=f2s[pick]),
            ),
        )
    end

    term = if mul_idx !== nothing && !(isapprox(c, 1.0; atol=1e-8))
        # scale the basis by the fitted coefficient
        constructorof(N)(;
            op=mul_idx, children=(constructorof(N)(T; val=T(c)), basis)
        )
    else
        basis  # no `*` operator available (or coefficient ≈ 1): add unscaled
    end

    newroot = constructorof(N)(; op=plus_idx, children=(tree, term))

    # Optionally add the intercept if it matters and we still have size budget.
    added = count_nodes(term) + 1
    if abs(b) > 0.05 * sr && base_size + added + 2 <= maxsize
        newroot = constructorof(N)(;
            op=plus_idx, children=(newroot, constructorof(N)(T; val=T(b)))
        )
    end

    return newroot  # new root!
end
```

## Parent mutation operator 2
```julia
"""
    symmetric_motif_duplication_gen4_9(tree::N, options, nfeatures::Int, rng::AbstractRNG) where {T,N<:AbstractExpressionNode{T}}

A structural mutation operator that identifies, replicates, and recombines modular subexpressions
(motifs) with optional feature-space permutations or direct coupling.

# Motivation & Core Idea
Physical and mathematical expressions commonly exhibit symmetries, repeated structural motifs
(e.g., Euclidean distances `(x1 - x2)^2 + (y1 - y2)^2`, coupled harmonic terms `sin(x1) + sin(x2)`,
or rational envelopes `u / (1 + u^2)`), and multilinear interactions. Standard point mutations
require exponentially many lucky mutations to replicate an evolved subexpression across other
variables. This operator extracts an existing subtree, systematically permutes its feature indices
(or keeps them identical for self-interaction), and couples or grafts the transformed motif back
into the expression tree.

# Steps & Heuristics
1. Sample a donor subtree from the existing expression tree.
2. Deep-copy the donor into an independent motif.
3. If features are present and `nfeatures > 1`, apply one of three feature-space transformations:
   - Cyclic feature permutation (e.g., `x_i -> x_{i+k}`), preserving relational sub-structures.
   - Targeted single-feature substitution (`x_a -> x_b`), useful for asymmetric couplings.
   - Identity mapping (no change), enabling polynomial powers (e.g. `u * u`) and rational forms.
4. Integrate the transformed motif into the tree via either:
   - Binary operator coupling: wraps a target node and the motif under a random binary operator.
   - Homologous grafting: replaces a target node in the tree with the evolved motif.
"""
function symmetric_motif_duplication_gen4_9(
    tree::N,
    options,
    nfeatures::Int,
    rng::AbstractRNG,
) where {T,N<:AbstractExpressionNode{T}}
    # Local helper: recursively collect all active feature indices in a subtree
    function _collect_features!(node, feats::Vector{Int})
        if node.degree == 0
            if !node.constant
                push!(feats, node.feature)
            end
        else
            for i in 1:(node.degree)
                _collect_features!(get_child(node, i), feats)
            end
        end
        return feats
    end

    # Local helper: apply a cyclic shift to all feature leaves in a subtree
    function _shift_features!(node, shift::Int, num_features::Int)
        if node.degree == 0
            if !node.constant
                node.feature = mod1(node.feature + shift, num_features)
            end
        else
            for i in 1:(node.degree)
                _shift_features!(get_child(node, i), shift, num_features)
            end
        end
        return node
    end

    # Local helper: substitute occurrences of a specific feature index with another
    function _remap_feature!(node, from_f::Int, to_f::Int)
        if node.degree == 0
            if !node.constant && node.feature == from_f
                node.feature = to_f
            end
        else
            for i in 1:(node.degree)
                _remap_feature!(get_child(node, i), from_f, to_f)
            end
        end
        return node
    end

    # Fallback: if no binary operators are configured, wrap with a unary operator if possible
    if options.nops[2] == 0
        if options.nops[1] > 0
            node = rand(rng, NodeSampler(; tree))
            new_node = constructorof(N)(;
                op=rand(rng, 1:options.nops[1]),
                children=(copy(node),),
            )
            set_node!(node, new_node)
        end
        return tree
    end

    # Step 1: Select a donor subtree motif to duplicate
    donor = rand(rng, NodeSampler(; tree))
    donor_clone = copy(donor)

    # Step 2: Extract active features and apply feature-space symmetry transformation
    feats = _collect_features!(donor_clone, Int[])
    if !isempty(feats) && nfeatures > 1
        strategy_roll = rand(rng)
        if strategy_roll < 0.45
            # Cyclic permutation: shifts all variable indices by a fixed offset
            shift = rand(rng, 1:(nfeatures - 1))
            _shift_features!(donor_clone, shift, nfeatures)
        elseif strategy_roll < 0.70
            # Targeted substitution: pick one variable present in the motif and remap it
            f_old = rand(rng, feats)
            valid_targets = [f for f in 1:nfeatures if f != f_old]
            if !isempty(valid_targets)
                f_new = rand(rng, valid_targets)
                _remap_feature!(donor_clone, f_old, f_new)
            end
        else
            # Identity: keep exact features to foster powers (e.g. u*u) or rational self-coupling
        end
    end

    # Step 3: Integrate the transformed motif back into the expression
    total_nodes = count_nodes(tree)
    # Use grafting if the tree has sufficient depth/size, otherwise couple via binary operator
    if total_nodes > 3 && rand(rng) < 0.35
        # Homologous Grafting: replace a randomly chosen node with the evolved motif
        target = rand(rng, NodeSampler(; tree))
        set_node!(target, donor_clone)
    else
        # Binary Coupling: pair target node with the motif under a binary operator
        target = rand(rng, Bool) ? donor : rand(rng, NodeSampler(; tree))
        target_copy = copy(target)
        bin_op = rand(rng, 1:options.nops[2])

        # Randomize operand ordering for non-commutative operations (subtraction, division)
        children_tuple = rand(rng, Bool) ? (target_copy, donor_clone) : (donor_clone, target_copy)
        new_node = constructorof(N)(;
            op=bin_op,
            children=children_tuple,
        )
        set_node!(target, new_node)
    end

    return tree
end
```
