"""
    affine_shape_relative_uniformity_loss_gen12_5(tree, dataset, options)

Shape-first loss that grades an expression by how well the *best affine calibration*
`a * prediction + b` reproduces the target, then refines that signal with two
uniformity-of-fit measures and a bounded raw-error tie-breaker.

Synthesis and motivation
------------------------
Both parents measure functional shape through the residual of the least-squares
affine calibration (an "angle" between prediction and target profiles), computed with
Welford moments in a first pass and with *directly evaluated fitted residuals* in a
second pass (avoiding the catastrophic cancellation of `var(y) - cov^2/var(p)` near an
exact structural match), and they report it as a normalized RMSE so that near-perfect
and exact structures are still distinguishable. A small saturating raw-error term then
selects the correctly calibrated representative without letting badly initialized
constants dominate shape discovery.

The observed failure mode of that pure variance-normalized criterion is visible in
searches such as `x0 / sqrt(1 - x1^2/x2^2)`: rational/Padé-like surrogates
(`x0 + 0.497*x0/(( x2/x1)^2 - 0.79)`) reach NMSE ~1e-4 and then monopolize the
frontier, because a global magnitude-weighted RMSE is nearly blind to *where* the
残 residual lives. Exact structure differs from such surrogates mainly in a limited
region (small denominators / small targets), i.e. in **relative**, not absolute, error.

This operator therefore keeps the affine shape RMSE as the primary term and adds:

  1. a *scale-damped relative* residual RMSE of the same affine-calibrated fit,
     `(fitted - y) / (|y| + target_rms/8)`, which weights every sample comparably and
     so rewards structures that are right across the whole range rather than only where
     `|y|` is large; the damping floor keeps it finite for targets crossing zero and
     prevents a handful of tiny-`|y|` points from taking over;
  2. a bounded *peak* relative residual, which penalizes locally-bad surrogates that
     look excellent on average — exactly the Padé-style plateau above;
  3. the parent's saturating raw-error term (weight 1/256) so that only genuinely
     calibrated predictions can reach zero loss.

All added terms use the affine-calibrated fit, so the criterion stays invariant to
overall scale/offset misfit of constants (the key property that lets structurally
correct but poorly tuned candidates survive), and all terms are bounded in `[0, 1]`,
so no single component can explode. Cost is one tree evaluation plus two cheap linear
passes with Kahan compensation — comparable to MSE. No complexity penalty is applied.
"""
function affine_shape_relative_uniformity_loss_gen12_5(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}
    prediction, completed = eval_tree_array(tree, dataset.X, options)
    if !completed || isnothing(prediction)
        return L(Inf)
    end

    n = dataset.n
    if n <= 0 || length(prediction) != n || length(dataset.y) != n
        return L(Inf)
    end

    zero_l = zero(L)
    one_l = one(L)
    n_l = L(n)

    # ---------------------------------------------------------------------
    # First pass: numerically stable Welford moments for the affine fit.
    # ---------------------------------------------------------------------
    mean_p = zero_l
    mean_y = zero_l
    m2_p = zero_l
    m2_y = zero_l
    cov_py = zero_l
    max_abs_p = zero_l
    max_abs_y = zero_l

    @inbounds for i in 1:n
        p = L(prediction[i])
        y = L(dataset.y[i])

        if !(isfinite(p) && isfinite(y))
            return L(Inf)
        end

        max_abs_p = max(max_abs_p, abs(p))
        max_abs_y = max(max_abs_y, abs(y))

        k = L(i)
        delta_p = p - mean_p
        delta_y = y - mean_y

        mean_p += delta_p / k
        mean_y += delta_y / k

        # Covariance update uses the freshly updated target mean.
        cov_py += delta_p * (y - mean_y)
        m2_p += delta_p * (p - mean_p)
        m2_y += delta_y * (y - mean_y)
    end

    if !(isfinite(mean_p) && isfinite(mean_y) &&
         isfinite(m2_p) && isfinite(m2_y) && isfinite(cov_py))
        return L(Inf)
    end

    # Welford sums of squares can go microscopically negative after roundoff.
    m2_p = max(m2_p, zero_l)
    m2_y = max(m2_y, zero_l)

    # Scale-relative variability tests remain meaningful for tiny datasets.
    relative_tolerance = sqrt(eps(L))
    target_rms = sqrt(m2_y / n_l)
    prediction_rms = sqrt(m2_p / n_l)
    target_is_variable = target_rms > relative_tolerance * max_abs_y
    prediction_is_variable = prediction_rms > relative_tolerance * max_abs_p

    if !target_is_variable
        # Affine shape is undefined for a constant target: fall back to a
        # scale-normalized raw MSE (normalize before squaring to avoid overflow).
        target_scale = max(one_l, max_abs_y)
        raw_sum = zero_l
        raw_compensation = zero_l

        @inbounds for i in 1:n
            p = L(prediction[i])
            y = L(dataset.y[i])
            residual = p - y
            if !isfinite(residual)
                return L(Inf)
            end

            normalized_residual = residual / target_scale
            term = normalized_residual * normalized_residual
            if !isfinite(term)
                return L(Inf)
            end

            corrected = term - raw_compensation
            updated = raw_sum + corrected
            if !isfinite(updated)
                return L(Inf)
            end
            raw_compensation = (updated - raw_sum) - corrected
            raw_sum = updated
        end

        raw_nmse = max(raw_sum / n_l, zero_l)
        loss_value = log1p(raw_nmse)
        return isfinite(loss_value) ? L(loss_value) : L(Inf)
    end

    target_norm = sqrt(m2_y)  # = sqrt(n) * target_rms
    if !(isfinite(target_norm) && target_norm > zero_l)
        return L(Inf)
    end

    # Least-squares affine calibration; a constant prediction collapses to mean(y).
    slope = zero_l
    intercept = mean_y
    if prediction_is_variable
        slope = cov_py / m2_p
        intercept = mean_y - slope * mean_p
        if !(isfinite(slope) && isfinite(intercept))
            return L(Inf)
        end
    end

    # Damping floor for the relative denominator: one eighth of the typical target
    # magnitude. Keeps near-zero targets from dominating while still letting the
    # small-|y| region contribute meaningfully.
    rel_floor = target_rms / L(8)

    # ---------------------------------------------------------------------
    # Second pass: fitted residuals (absolute, relative, peak) and raw residuals.
    # ---------------------------------------------------------------------
    affine_sum = zero_l
    affine_compensation = zero_l
    rel_sum = zero_l
    rel_compensation = zero_l
    max_rel_sq = zero_l
    raw_sum = zero_l
    raw_compensation = zero_l
    raw_overflowed = false
    rel_overflowed = false

    @inbounds for i in 1:n
        p = L(prediction[i])
        y = L(dataset.y[i])

        raw_residual = p - y
        if !isfinite(raw_residual)
            return L(Inf)
        end

        # --- bounded raw calibration term (normalized by ||y - mean(y)||) ---
        if !raw_overflowed
            raw_normalized = raw_residual / target_norm
            raw_term = raw_normalized * raw_normalized
            if isfinite(raw_term)
                corrected = raw_term - raw_compensation
                updated = raw_sum + corrected
                if isfinite(updated)
                    raw_compensation = (updated - raw_sum) - corrected
                    raw_sum = updated
                else
                    raw_overflowed = true
                end
            else
                raw_overflowed = true
            end
        end

        # --- affine-calibrated residual (primary shape signal) ---
        fitted = prediction_is_variable ? muladd(slope, p, intercept) : mean_y
        affine_residual = fitted - y
        affine_normalized = affine_residual / target_norm
        affine_term = affine_normalized * affine_normalized

        if !(isfinite(fitted) && isfinite(affine_residual) &&
             isfinite(affine_normalized) && isfinite(affine_term))
            return L(Inf)
        end

        corrected = affine_term - affine_compensation
        updated = affine_sum + corrected
        if !isfinite(updated)
            return L(Inf)
        end
        affine_compensation = (updated - affine_sum) - corrected
        affine_sum = updated

        # --- scale-damped relative residual of the same calibrated fit ---
        if !rel_overflowed
            rel_denominator = abs(y) + rel_floor
            rel_residual = affine_residual / rel_denominator
            rel_term = rel_residual * rel_residual
            if isfinite(rel_term)
                max_rel_sq = max(max_rel_sq, rel_term)
                corrected_rel = rel_term - rel_compensation
                updated_rel = rel_sum + corrected_rel
                if isfinite(updated_rel)
                    rel_compensation = (updated_rel - rel_sum) - corrected_rel
                    rel_sum = updated_rel
                else
                    rel_overflowed = true
                end
            else
                rel_overflowed = true
            end
        end
    end

    # The affine optimum can never be worse than predicting mean(y), i.e. NMSE <= 1.
    affine_nmse = clamp(affine_sum, zero_l, one_l)
    shape_term = sqrt(affine_nmse)  # RMSE-like: keeps resolution near exactness

    # Mean relative accuracy, squashed into [0, 1) then square-rooted so that it
    # behaves like a relative RMSE for small errors (fine gradient near exactness).
    rel_term_value = if rel_overflowed
        one_l
    else
        rel_mse = max(rel_sum / n_l, zero_l)
        sqrt(rel_mse / (one_l + rel_mse))
    end

    # Worst-case (peak) relative deviation: punishes surrogates that are excellent
    # on average but structurally wrong in a localized region.
    peak_term_value = if rel_overflowed
        one_l
    else
        peak_rel = sqrt(max(max_rel_sq, zero_l))
        peak_rel / (one_l + peak_rel)
    end

    # Bounded raw error: monotone preference for the correctly calibrated member,
    # saturating so that mis-scaled but structurally right candidates survive.
    bounded_raw = if raw_overflowed
        one_l
    else
        raw_nmse = max(raw_sum, zero_l)
        raw_nmse <= one_l ?
            raw_nmse / (one_l + raw_nmse) :
            one_l / (one_l + one_l / raw_nmse)
    end

    # Weights: shape dominates; relative uniformity is a strong secondary signal;
    # peak deviation is a mild regularizer; raw calibration is only a tie-breaker.
    relative_weight = one_l / L(4)
    peak_weight = one_l / L(16)
    calibration_weight = one_l / L(256)

    loss_value =
        shape_term +
        relative_weight * rel_term_value +
        peak_weight * peak_term_value +
        calibration_weight * bounded_raw

    return isfinite(loss_value) ? L(max(loss_value, zero_l)) : L(Inf)
end

"""
    affine_angular_profile_loss_2pass_gen8_1(tree, dataset, options)

Measure functional shape with the residual angle of the best affine calibration
`a * prediction + b`, then use a small bounded raw-error term to select the correctly
calibrated representative. The first pass computes stable Welford moments and
scale-aware variability tests. A second pass evaluates the fitted residuals directly,
avoiding the cancellation in `var(y) - cov(p,y)^2 / var(p)` near an exact structural
match.

Compared with the parent, the affine residual is expressed as normalized RMSE rather
than log-NMSE, magnifying distinctions between very accurate approximations and exact
structure. Raw error is retained as a bounded calibration tie-breaker, so only the
actual target predictions attain zero while poorly tuned constants cannot dominate
shape discovery. Variability thresholds are scale-relative, constant targets fall
back to normalized raw error, and no expression-complexity penalty is included.
"""
function affine_angular_profile_loss_2pass_gen8_1(
    tree::Union{AbstractExpression{T},AbstractExpressionNode{T}},
    dataset::Dataset{T,L},
    options::AbstractOptions,
)::L where {T<:DATA_TYPE,L<:LOSS_TYPE}

        return L(Inf)
    end

    one_l = one(L)
    n_l = L(n)

    mean_y = zero_l
    m2_p = zero_l
    m2_y = zero_l

        max_abs_y = max(max_abs_y, abs(y))

        mean_y += delta_y / k

        m2_p += delta_p * (p - mean_p)
        m2_y += delta_y * (y - mean_y)
    end

    if !(isfinite(mean_p) && isfinite(mean_y) &&
         isfinite(m2_p) && isfinite(m2_y) && isfinite(cov_py))
        return L(Inf)
    end

    # Welford sums can be microscopically negative after roundoff.
    m2_p = max(m2_p, zero_l)
    m2_y = max(m2_y, zero_l)

    # Relative standard-deviation tests remain meaningful for very small data.
    relative_tolerance = sqrt(eps(L))
    target_rms = sqrt(m2_y / n_l)
    prediction_rms = sqrt(m2_p / n_l)
    target_is_variable =
        target_rms > relative_tolerance * max_abs_y
    prediction_is_variable =
        prediction_rms > relative_tolerance * max_abs_p

    if !target_is_variable
        # Affine shape is undefined for a constant target. Use scale-normalized
        # raw MSE, accumulated after division to avoid squaring large targets.
        target_scale = max(one_l, max_abs_y)
        raw_sum = zero_l
        raw_compensation = zero_l

        @inbounds for i in 1:n
            p = L(prediction[i])
            y = L(dataset.y[i])
            residual = p - y

            if !isfinite(residual)
                return L(Inf)
            end

            normalized_residual = residual / target_scale
            term = normalized_residual * normalized_residual
            if !isfinite(term)
                return L(Inf)
            end

            corrected = term - raw_compensation
            updated = raw_sum + corrected
            if !isfinite(updated)
                return L(Inf)
            end
            raw_compensation = (updated - raw_sum) - corrected
            raw_sum = updated
        end

        raw_nmse = max(raw_sum / n_l, zero_l)
        loss_value = log1p(raw_nmse)
        return isfinite(loss_value) ? L(loss_value) : L(Inf)
    end

    target_norm = sqrt(m2_y)
    if !(isfinite(target_norm) && target_norm > zero_l)
        return L(Inf)
    end

    slope = zero_l
    intercept = mean_y
    if prediction_is_variable
        slope = cov_py / m2_p
        intercept = mean_y - slope * mean_p
        if !(isfinite(slope) && isfinite(intercept))
            return L(Inf)
        end
    end

    # Second pass: directly evaluate the fitted residual. This is more accurate
    # near perfect correlation than subtracting two almost equal variances.
    affine_sum = zero_l
    affine_compensation = zero_l
    raw_sum = zero_l
    raw_compensation = zero_l
    raw_overflowed = false

    @inbounds for i in 1:n
        p = L(prediction[i])
        y = L(dataset.y[i])

        raw_residual = p - y
        if !isfinite(raw_residual)
            return L(Inf)
        end

        if !raw_overflowed
            raw_normalized = raw_residual / target_norm
            raw_term = raw_normalized * raw_normalized

            if isfinite(raw_term)
                corrected = raw_term - raw_compensation
                updated = raw_sum + corrected
                if isfinite(updated)
                    raw_compensation = (updated - raw_sum) - corrected
                    raw_sum = updated
                else
                    raw_overflowed = true
                end
            else
                raw_overflowed = true
            end
        end

        fitted = prediction_is_variable ? muladd(slope, p, intercept) : mean_y
        affine_residual = fitted - y
        affine_normalized = affine_residual / target_norm
        affine_term = affine_normalized * affine_normalized

        if !(isfinite(fitted) && isfinite(affine_residual) &&
             isfinite(affine_normalized) && isfinite(affine_term))
            return L(Inf)
        end

        updated = affine_sum + corrected
        if !isfinite(updated)
            return L(Inf)
        end

        affine_sum = updated
    end

    affine_nmse = clamp(affine_sum, zero_l, one_l)
    shape_term = sqrt(affine_nmse)

    # Saturation lets affine-equivalent structures compete despite poor initial
    # constants, while preserving a monotone preference for raw calibration.
    bounded_raw = if raw_overflowed
        one_l
    else

        raw_nmse <= one_l ?
            raw_nmse / (one_l + raw_nmse) :
            one_l / (one_l + one_l / raw_nmse)
    end

    calibration_weight = one_l / L(256)
    loss_value = shape_term + calibration_weight * bounded_raw

    return isfinite(loss_value) ? L(max(loss_value, zero_l)) : L(Inf)
end
