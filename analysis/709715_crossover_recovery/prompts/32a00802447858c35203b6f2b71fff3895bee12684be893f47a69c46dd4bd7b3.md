## Parent selection operator 1
```julia
"""
    crowding_suppressed_niche_tournament_gen10_4(pop, running_search_statistics, options)

Tournament selection that actively fights the two failure modes most visible in
long PySR runs: **lineage monopoly** (one champion skeleton and its near-identical
clones fill the population, so search degenerates into constant-tuning of a single
structure) and **complexity-niche starvation** (whole complexity bands go extinct,
removing the mid-size building blocks needed to assemble the true expression).

Core idea: score a slightly enlarged tournament pool with the usual
frequency-adjusted cost, then *inflate* the score of members that are redundant
copies of a better pool member (same complexity + essentially the same loss, or a
shared parent reference), *slightly discount* the freshest members to keep lineage
turnover going, and occasionally (with probability `p_rare_niche`) deliberately
select the pool member sitting in the rarest complexity niche instead of the
fittest one.

Steps:
 1. Sample a pool of `~1.5 * tournament_selection_n` distinct members (a bit larger
    than the default tournament, so that redundancy/niche information exists).
 2. Compute complexity and the default adaptive-parsimony-adjusted cost
    (`cost * exp(scaling * normalized_frequency[complexity])`).
 3. Redundancy (crowding) suppression: sort the pool by adjusted cost; for each
    member count how many *better* pool members are effectively the same solution
    (identical complexity and loss within a relative tolerance -> weight 1.0) or a
    sibling from the same parent (-> weight `sibling_weight`). Inflate the score
    geometrically with that accumulated redundancy weight. The best representative
    of each cluster is untouched, its twins are progressively demoted.
 4. Youth discount: members born most recently get up to `youth_discount`
    relative cost reduction, which prevents an ancient champion from being
    re-selected forever and re-tuned in place.
 5. With probability `p_rare_niche`, return the pool member whose complexity has
    the lowest `normalized_frequencies` value (ties broken by score) — an explicit
    exploration move that repopulates starved complexity bands. Otherwise run the
    standard geometric tournament ladder (`tournament_selection_p`) over the
    penalized scores.

Assumptions / heuristics:
 - Equal complexity plus (relatively) equal loss is a cheap, allocation-free proxy
   for "structurally the same expression"; false positives merely nudge selection
   toward a slightly different equally-good member, which is harmless.
 - Costs are usually positive; the `inflate` helper handles non-positive costs by
   dividing instead of multiplying so that "worse" always means "worse".
 - All penalties are multiplicative and mild, so behaviour degrades gracefully to
   the default tournament when the pool contains no redundancy.
"""
function crowding_suppressed_niche_tournament_gen10_4(
    pop::Population{T,L,N},
    running_search_statistics::RunningSearchStatistics,
    options::AbstractOptions,
)::PopMember{T,L,N} where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    # ---------------- tunables (kept mild so we degrade to default behaviour) ----
    pool_scale = 1.5          # pool size relative to tournament_selection_n
    clone_penalty = 1.35      # score inflation per unit of redundancy weight
    sibling_weight = 0.4      # a shared parent counts as a partial twin
    loss_rtol = 1e-6          # relative tolerance for "same loss"
    youth_discount = 0.05     # max relative cost discount for the newest members
    p_rare_niche = 0.10       # probability of the explicit rare-complexity pick

    # Multiplicative penalty/bonus that is sign-safe (f > 1 always means worse).
    inflate = function (x::L, f::Float64)
        fl = L(f)
        return x > zero(L) ? x * fl : x / fl
    end

    # ---------------- 1. sample the pool -----------------------------------------
    n_base = options.tournament_selection_n
    pool_size = min(pop.n, max(n_base, round(Int, pool_scale * n_base)))
    pool = StatsBase.sample(pop.members, pool_size; replace=false)
    n = length(pool)

    # ---------------- 2. base (frequency-adjusted) scores ------------------------
    complexities = Vector{Int}(undef, n)
    scores = Vector{L}(undef, n)
    adaptive_parsimony_scaling = L(options.adaptive_parsimony_scaling)
    for i in 1:n
        member = pool[i]
        size = compute_complexity(member, options)
        complexities[i] = size
        if options.use_frequency_in_tournament
            frequency = if (0 < size <= options.maxsize)
                L(running_search_statistics.normalized_frequencies[size])
            else
                L(0)
            end
            scores[i] = member.cost * exp(adaptive_parsimony_scaling * frequency)
        else
            scores[i] = member.cost
        end
    end

    # ---------------- 3. redundancy (clone / sibling) suppression -----------------
    # Process from best to worst so the best representative of a cluster is free of
    # penalty and each of its twins gets progressively demoted.
    order = sortperm(scores)
    redundancy = zeros(Float64, n)
    for a in 2:n
        i = order[a]
        li = pool[i].loss
        wi = 0.0
        for b in 1:(a - 1)
            j = order[b]
            lj = pool[j].loss
            # near-identical solution: same complexity and (relatively) same loss
            same_loss = abs(li - lj) <= L(loss_rtol) * max(abs(li), abs(lj), eps(L))
            if complexities[i] == complexities[j] && same_loss
                wi += 1.0
            elseif pool[i].parent != 0 && pool[i].parent == pool[j].parent
                # siblings from the same parent: partial redundancy
                wi += sibling_weight
            end
        end
        redundancy[i] = wi
    end

    # ---------------- 4. youth discount ------------------------------------------
    birth_min = pool[1].birth
    birth_max = pool[1].birth
    for i in 2:n
        b = pool[i].birth
        b < birth_min && (birth_min = b)
        b > birth_max && (birth_max = b)
    end
    birth_span = max(birth_max - birth_min, 1)

    penalized = Vector{L}(undef, n)
    for i in 1:n
        # redundancy inflation (>= 1) and recency discount (<= 1) combined
        recency = (pool[i].birth - birth_min) / birth_span  # 0 (oldest) .. 1 (newest)
        factor = (clone_penalty^redundancy[i]) * (1.0 - youth_discount * recency)
        penalized[i] = inflate(scores[i], factor)
    end

    # ---------------- 5a. occasional rare-complexity (niche) pick -----------------
    if rand() < p_rare_niche
        best_idx = 1
        best_freq = Inf
        best_score = penalized[1]
        for i in 1:n
            c = complexities[i]
            f = if (0 < c <= options.maxsize)
                Float64(running_search_statistics.normalized_frequencies[c])
            else
                0.0  # out-of-range sizes count as maximally rare
            end
            # prefer the rarest complexity; break ties with the penalized score
            if f < best_freq || (f == best_freq && penalized[i] < best_score)
                best_freq = f
                best_score = penalized[i]
                best_idx = i
            end
        end
        return pool[best_idx]
    end

    # ---------------- 5b. standard geometric tournament ladder --------------------
    p = options.tournament_selection_p
    chosen_idx = if p == 1.0
        argmin_fast(penalized)
    else
        k = collect(0:(n - 1))
        prob_each = p * ((1 - p) .^ k)
        weights = StatsBase.Weights(prob_each, sum(prob_each))
        tournament_winner = StatsBase.sample(weights)
        if tournament_winner == 1
            argmin_fast(penalized)
        else
            bottomk_fast(penalized, tournament_winner)[2][end]
        end
    end
    return pool[chosen_idx]
end
```
