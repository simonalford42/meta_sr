```julia
"""
    pareto_crowded_age_survival_init_4(
        pop::Population{T,L,N},
        options::AbstractOptions;
        exclude_indices::Vector{Int}=Int[],
    )::Int where {T,L,N}

Multi-objective survival operator for symbolic regression that balances age-regularized
evolutionary turnover, Pareto frontier preservation, and complexity-niche crowding penalty.

# Core Idea
Standard regularized evolution evicts the oldest individual regardless of whether it is a
breakthrough Pareto-optimal solution or a stagnant, heavily dominated duplicate.
This operator computes an eviction priority score for each candidate member by synthesizing
three evolutionary dynamics:
1. **Age Regularization (Exploration turnover)**: Older individuals receive higher eviction
   priority to ensure continuous turnover of genetic material.
2. **Pareto Dominance Shielding (Stepping-stone preservation)**: Solutions on or near the
   empirical Pareto front (Loss vs. Complexity) receive strong protection against premature
   eviction, preserving non-dominated stepping stones across the search space.
3. **Complexity-Niche Crowding & De-duplication**: Individuals that are dominated within
   heavily crowded complexity bins (multiple individuals at the identical complexity) are
   penalized and prioritized for eviction, freeing capacity for novel complexity niches.
4. **Immediate Sanitation**: Individuals with invalid (NaN/Inf) loss or cost are evicted
   immediately before any valid population member.

When all individuals in the population are non-dominated and uniquely distributed across
complexity bins, the dominance and crowding penalties become zero, and the operator smoothly
reduces to classic Age-Regularized Evolution.

# Returns
- `Int`: The 1-based index of the population member selected for replacement.
"""
function pareto_crowded_age_survival_init_4(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T,L,N}
    n = pop.n
    n <= 0 && return 1

    # Find first eligible index as a safe fallback for edge cases
    fallback_idx = 1
    for i in 1:n
        if !(i in exclude_indices)
            fallback_idx = i
            break
        end
    end

    # Pre-extract properties for fast vector processing
    complexities = Vector{Int}(undef, n)
    losses = Vector{Float64}(undef, n)
    births = Vector{Int}(undef, n)
    is_invalid = Vector{Bool}(undef, n)

    min_birth = typemax(Int)
    max_birth = typemin(Int)

    for i in 1:n
        member = pop.members[i]
        comp = compute_complexity(member, options)
        loss = Float64(member.loss)
        cost = Float64(member.cost)
        birth = member.birth

        complexities[i] = comp
        losses[i] = loss
        births[i] = birth

        # Check for non-finite / invalid numerical values
        invalid = isnan(loss) || isinf(loss) || isnan(cost) || isinf(cost)
        is_invalid[i] = invalid

        if !(i in exclude_indices)
            # Immediate eviction for non-finite solutions to sanitize the population
            invalid && return i

            if birth < min_birth
                min_birth = birth
            end
            if birth > max_birth
                max_birth = birth
            end
        end
    end

    # If all valid members are excluded or population is degenerate
    if min_birth > max_birth
        return fallback_idx
    end

    birth_span = max(1, max_birth - min_birth)

    # Compute Pareto Dominance count: how many valid members strictly dominate member i
    # Member j dominates member i if:
    #   (comp_j <= comp_i && loss_j <= loss_i) && (comp_j < comp_i || loss_j < loss_i)
    dom_counts = zeros(Int, n)
    max_dom = 0
    for i in 1:n
        (i in exclude_indices || is_invalid[i]) && continue
        comp_i = complexities[i]
        loss_i = losses[i]
        d_count = 0
        for j in 1:n
            (j == i || is_invalid[j]) && continue
            comp_j = complexities[j]
            loss_j = losses[j]
            if (comp_j <= comp_i && loss_j <= loss_i) && (comp_j < comp_i || loss_j < loss_i)
                d_count += 1
            end
        end
        dom_counts[i] = d_count
        if d_count > max_dom
            max_dom = d_count
        end
    end

    # Compute Complexity-Niche Crowding & Intra-bin Loss Rank:
    # Measures how crowded member i's complexity bin is and how sub-optimal member i is within it
    crowding_scores = zeros(Float64, n)
    for i in 1:n
        (i in exclude_indices || is_invalid[i]) && continue
        comp_i = complexities[i]
        loss_i = losses[i]

        bin_count = 0
        worse_in_bin_count = 0

        for j in 1:n
            is_invalid[j] && continue
            if complexities[j] == comp_i
                bin_count += 1
                if losses[j] < loss_i
                    worse_in_bin_count += 1
                end
            end
        end

        # Crowding penalty is non-zero only if multiple individuals share the same complexity
        if bin_count > 1
            relative_rank = worse_in_bin_count / (bin_count - 1)
            density_factor = bin_count / n
            crowding_scores[i] = relative_rank * density_factor
        end
    end

    # Compute composite eviction score:
    # Higher score -> higher priority for replacement
    best_eviction_score = -Inf
    best_idx = fallback_idx

    # Weight parameters for multi-objective eviction
    w_age = 0.40      # Base aging pressure for exploration turnover
    w_dom = 0.35      # Pareto dominance penalty (dominated solutions get evicted first)
    w_crowd = 0.25    # Complexity niche crowding penalty (prunes redundant bin duplicates)

    for i in 1:n
        (i in exclude_indices) && continue

        # Normalized age: 1.0 for oldest, 0.0 for youngest
        norm_age = (max_birth - births[i]) / birth_span

        # Normalized dominance: 0.0 for Pareto-optimal non-dominated solutions
        norm_dom = max_dom > 0 ? (dom_counts[i] / max_dom) : 0.0

        # Crowding score: 0.0 for bin champions or unique complexity bins
        norm_crowd = crowding_scores[i]

        eviction_score = w_age * norm_age + w_dom * norm_dom + w_crowd * norm_crowd

        if eviction_score > best_eviction_score
            best_eviction_score = eviction_score
            best_idx = i
        end
    end

    return best_idx
end
```

## Parent survival operator 2
```julia
"""
    pareto_niche_crowding_survival_init_3(pop, options; exclude_indices=Int[])

Survival (replacement) operator that combines **age-regularization** with
**Pareto-front protection** and **complexity-niche crowding pressure**.

Core idea
---------
The default operator always kills the oldest member. That maintains turnover, but it
happily destroys the *only* accurate expression at some complexity level, which is
exactly the kind of stepping stone symbolic regression needs to reach a ground-truth
formula. Conversely, purely greedy replacement (kill the worst) collapses diversity.

This operator instead ranks every member by an *expendability score* built from four
signals, and replaces the most expendable one:

1. **Dominance count** — how many members are at least as simple *and* at least as
   accurate. A heavily dominated member carries no information the population does not
   already have, so it is the prime candidate for replacement.
2. **Structural-clone count** — members with identical complexity and (numerically)
   identical loss are duplicates; all but the oldest representative are pure waste.
   These are penalized hardest because clones are the main driver of premature
   convergence in tournament-based SR.
3. **Niche crowding** — how many other members occupy the same complexity level.
   Replacement pressure is concentrated on over-populated complexity bins, which keeps
   the population spread across the whole complexity axis (a proxy for the Pareto
   "hall of fame" that PySR ultimately reports).
4. **Age** — older members get a mild bonus to expendability, preserving the
   exploration benefits of age-regularized evolution.

Protections
-----------
* Non-dominated, non-duplicated members (the population's own Pareto front) receive a
  large score discount, so they are only replaced when nothing else is available.
* The single best-cost member is effectively never replaced (elitism).
* Members with non-finite loss/cost are treated as maximally expendable.

Stochastic age branch
---------------------
With probability `p_age` the operator falls back to pure oldest-first replacement
(restricted to unprotected members). This guarantees that no member can survive
forever by sitting in a sparse niche, avoiding the stagnation that fully
deterministic greedy/crowding schemes suffer from.

Assumptions / heuristics
------------------------
* `(complexity, loss)` is used for dominance (rather than `cost`) so that the operator's
  notion of "redundant" is independent of the adaptive parsimony term, which fluctuates.
* Complexity bins are exact integer complexities; this is a natural niche definition
  because SR complexity is already a small integer.
* Small random noise breaks score ties uniformly, so symmetric duplicates do not always
  lose the same slot.
* `exclude_indices` entries are given score `-Inf` and can therefore never be returned.
"""
function pareto_niche_crowding_survival_init_3(
    pop::Population{T,L,N},
    options::AbstractOptions;
    exclude_indices::Vector{Int}=Int[],
)::Int where {T<:DATA_TYPE,L<:LOSS_TYPE,N}
    n = pop.n

    # Trivial case: single-member population.
    n == 1 && return 1

    # ---------------------------------------------------------------------
    # 1. Gather per-member statistics once (complexity, loss, cost, birth).
    # ---------------------------------------------------------------------
    complexities = Vector{Int}(undef, n)
    losses = Vector{Float64}(undef, n)
    costs = Vector{Float64}(undef, n)
    births = Vector{Float64}(undef, n)
    for i in 1:n
        m = pop.members[i]
        complexities[i] = compute_complexity(m, options)
        l = Float64(m.loss)
        # Non-finite (NaN/Inf) losses are mapped to +Inf: worst possible, fully dominated.
        losses[i] = isfinite(l) ? l : Inf
        c = Float64(m.cost)
        costs[i] = isfinite(c) ? c : Inf
        births[i] = Float64(m.birth)
    end

    min_birth, max_birth = extrema(births)
    birth_range = max(max_birth - min_birth, 1.0)  # guard against all-equal births

    # Elite (best cost) member: protected from replacement.
    best_cost_idx = 1
    for i in 2:n
        if costs[i] < costs[best_cost_idx]
            best_cost_idx = i
        end
    end

    # ---------------------------------------------------------------------
    # 2. Pairwise pass: dominance counts, clone counts, niche occupancy.
    # ---------------------------------------------------------------------
    n_dom = zeros(Int, n)            # number of members dominating i in (complexity, loss)
    n_clone = zeros(Int, n)          # number of older/earlier exact duplicates of i
    n_niche = ones(Int, n)           # members sharing i's complexity (including i itself)

    for i in 1:n
        ci = complexities[i]
        li = losses[i]
        bi = births[i]
        for j in 1:n
            j == i && continue
            cj = complexities[j]
            lj = losses[j]

            # Pareto dominance: simpler-or-equal and better-or-equal, strict in one.
            if (cj <= ci && lj < li) || (cj < ci && lj <= li)
                n_dom[i] += 1
            end

            if cj == ci
                n_niche[i] += 1
                # Numerically identical loss at identical complexity => structural clone.
                same_loss =
                    (isinf(li) && isinf(lj)) || abs(lj - li) <= 1e-10 * (1.0 + abs(li))
                if same_loss
                    # Keep the earliest-born representative; count only *earlier* twins,
                    # so exactly one member of a clone cluster ends up unpenalized.
                    if bj_is_earlier(births[j], bi, j, i)
                        n_clone[i] += 1
                    end
                end
            end
        end
    end

    # A member is "protected" if it sits on the population's Pareto front and is
    # not a redundant copy of an equally good, equally complex member.
    protected = [(n_dom[i] == 0 && n_clone[i] == 0) for i in 1:n]

    excluded = Set(exclude_indices)

    # ---------------------------------------------------------------------
    # 3. Stochastic pure-age branch (age-regularized evolution fallback).
    #    Keeps turnover alive so nothing survives indefinitely.
    # ---------------------------------------------------------------------
    p_age = 0.2
    if rand() < p_age
        best_idx = 0
        best_birth = Inf
        # First try unprotected, non-elite members only.
        for i in 1:n
            (i in excluded) && continue
            i == best_cost_idx && continue
            protected[i] && continue
            if births[i] < best_birth
                best_birth = births[i]
                best_idx = i
            end
        end
        best_idx != 0 && return best_idx
        # Otherwise fall through to the scoring path (which relaxes protections).
    end

    # ---------------------------------------------------------------------
    # 4. Expendability score: higher == more deserving of replacement.
    # ---------------------------------------------------------------------
    scores = Vector{Float64}(undef, n)
    for i in 1:n
        if i in excluded
            scores[i] = -Inf   # never returnable
            continue
        end

        s = 0.0

        # (a) Redundancy in the accuracy/complexity trade-off.
        s += 3.0 * log1p(n_dom[i])

        # (b) Exact duplicates: strongest signal, they add zero diversity.
        s += 4.0 * n_clone[i]

        # (c) Crowding within the member's complexity niche.
        s += 1.5 * log1p(max(n_niche[i] - 1, 0))

        # (d) Age bonus: oldest member of the population gets ~1.0, newest ~0.0.
        s += 1.0 * (1.0 - (births[i] - min_birth) / birth_range)

        # (e) Broken/degenerate members are always fair game.
        if !isfinite(losses[i]) || !isfinite(costs[i])
            s += 10.0
        end

        # (f) Oversized expressions beyond the allowed maxsize should not linger.
        if complexities[i] > options.maxsize
            s += 5.0
        end

        # Protections (applied as large discounts so they still lose to -Inf exclusions).
        if protected[i]
            s -= 50.0
        end
        if i == best_cost_idx
            s -= 1.0e6  # elitism: effectively never replace the best-cost member
        end

        # Random tie-break jitter, small relative to the structural terms.
        s += 0.05 * rand()

        scores[i] = s
    end

    # argmax over scores (equivalently argmin over negated scores).
    return argmin_fast([-scores[i] for i in 1:n])
end

"""
    bj_is_earlier(bj, bi, j, i)

Tie-break helper for clone detection: returns `true` when member `j` should be
considered the *earlier* (surviving) representative of a clone cluster, using birth
order first and index as a deterministic fallback for simultaneous births.
"""
function bj_is_earlier(bj::Float64, bi::Float64, j::Int, i::Int)
    return bj < bi || (bj == bi && j < i)
end
```
