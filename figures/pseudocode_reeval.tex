% Condensed meta-evolution focusing on reevaluation and MAP-Elites population.

\begin{algorithm}[H]
\caption{Meta-evolution}
\small
\textbf{Inputs:} baseline; $G$ generations; $O$ offspring per generation;
$P$ population size; tasks $T$;\\
$N_{\mathrm{init}}$ initial seeds; $B$ reevaluation seeds per generation;
$N_{\mathrm{val}}$ validation seeds.
\medskip
\begin{algorithmic}[1]
\State $\name{population} \gets [\name{baseline}]$
\State $\name{archive} \gets \{\}$
\For{$g = 1,\ldots,G$}
    \For{$o = 1,\ldots,O$} \Comment{Sample offspring}
        \State $\name{parents} \gets \Call{SampleParents}{\name{population}}$
        \State $\name{prompt} \sim \Call{Prompt}{T, g, \name{parents}}$
        \State $x_o \sim \Call{LLM}{\name{prompt}}$
        \State $\Call{Evaluate}{x_o, T, N_{\mathrm{init}}}$
        \State $\name{archive} \gets \name{archive}\cup\{x_o\}$
    \EndFor
    \State $\hat\mu_x, n_x \gets$ mean score, seed count of $x\in\name{archive}$ \Comment{Reevaluate with TTTS}
    \State $\psi_x \gets$ top-two Thompson probability of $x$ under $\mathcal{N}(\hat\mu_x, \hat\sigma^2/n_x)$
    \State Draw $B$ members from $\name{archive}$ with probability $\propto \psi_x$
    \State Evaluate each drawn member on new seeds and update $\hat\mu_x, n_x$
    \State $\name{population} \gets \{\arg\max_{x\in\name{archive}} \hat\mu_x(t) \mid t\in T\}$ \Comment{MAP-Elites, one niche per task}
    \State Fill remaining slots by highest overall $\hat\mu_x$
\EndFor
\For{$x\in\name{population}$} \Comment{Final validation reevaluation}
    \State $\name{val}[x] \gets \Call{Evaluate}{x, T_{\mathrm{val}}, N_{\mathrm{val}}}$
\EndFor
\State \Return $\arg\max_{x\in\name{population}} \name{val}[x]$
\end{algorithmic}
\end{algorithm}
