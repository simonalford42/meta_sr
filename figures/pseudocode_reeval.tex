% Condensed meta-evolution focusing on reevaluation and MAP-Elites population.

\begin{algorithm}[H]
\caption{Meta-evolution}
\small
\textbf{Inputs:} baseline; $G$ generations; $O$ offspring per generation;
$P$ population size; tasks $T$;\\
$N_{\mathrm{init}}$ initial seeds; $B$ TTTS seeds per generation;
$N_{\mathrm{reeval}}$ uniform reevaluation seeds; $N_{\mathrm{val}}$ validation seeds;\\
reevaluation $\name{strategy}\in\{\name{TTTS},\name{uniform}\}$.
\medskip
\begin{algorithmic}[1]
\State $\name{population} \gets [\name{baseline}]$
\State $\name{archive} \gets \{\}$
\For{$g = 1,\ldots,G$}
    \For{$o = 1,\ldots,O$} \Comment{Sample offspring}
        \State $\name{parents} \sim \Call{SampleParents}{\name{population}}$
        \State $\name{prompt} \sim \Call{Prompt}{T, g, \name{parents}}$
        \State $x_o \sim \Call{LLM}{\name{prompt}}$
        \State $\hat\mu_{x_o} \gets \Call{Evaluate}{x_o, T, N_{\mathrm{init}}}$
        \State $n_{x_o} \gets N_{\mathrm{init}}$
        \State $\name{archive} \gets \name{archive}\cup\{x_o\}$
    \EndFor
    \If{$\name{strategy} = \name{TTTS}$} \Comment{Reevaluate}
        \State $\psi_x \gets$ top-two Thompson probability of $x$ under $\mathcal{N}(\hat\mu_x, \hat\sigma^2/n_x)$
        \State Draw $B$ members from $\name{archive}$ with probability $\propto \psi_x$
    \ElsIf{$\name{strategy} = \name{uniform}$}
        \State Draw each new member of $\name{population}$ $N_{\mathrm{reeval}}$ times
    \EndIf
    \State Evaluate each drawn member on new seeds and update $\hat\mu_x$, $n_x$
    \State $\name{population} \gets \{\arg\max_{x\in\name{archive}} \hat\mu_x(t) \mid t\in T\}$ \Comment{MAP-Elites, one niche per task}
    \State $x^*_g \gets \arg\max_{x\in\name{population}} \hat\mu_x$ \Comment{Validate generation's best}
    \State $\name{val}[x^*_g] \gets \Call{Evaluate}{x^*_g, T_{\mathrm{val}}, N_{\mathrm{val}}}$
\EndFor
\State \Return $\arg\max_{x} \name{val}[x]$
\end{algorithmic}
\end{algorithm}
