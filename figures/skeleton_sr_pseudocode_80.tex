\paragraph{Overview.}
SkeletonSR is a minimal PySR-style symbolic-regression engine. The fixed
\emph{skeleton} (this document) handles tree representation, budgets,
random tree construction, constraints, constant optimisation and the
outer evolutionary loop. Everything that decides \emph{which} expressions
win is delegated to a pluggable \emph{policy} with the slots shown in
\textcolor{blue!70!black}{\textsc{blue small caps}}:
\slot{init\_state}, \slot{loss\_function}, \slot{selection},
\slot{mutation}, \slot{crossover}, \slot{acceptance}, \slot{survival},
\slot{update\_population}, \slot{update\_state!}, plus the optional
\slot{cycles\_per\_population} and \slot{should\_crossover}.

\section*{Data structures}
\begin{algorithmic}[1]
\State \textbf{Node} $\;::=\;$ ConstNode(value) $\mid$ VarNode(feature) $\mid$ OpNode(op, left, right$\mid\varnothing$)
\State \textbf{Individual} $\;=\;$ (tree, loss, cost, complexity, birth, ref, parent\_ref)
\State \textbf{EngineConfig} $\;=\;$ (population\_size, populations, niterations, ncycles\_per\_iteration, maxsize, maxdepth,
\Statex \hspace{2.2em} max\_evals, timeout, binary\_ops, unary\_ops, constants, constraints, nested\_constraints, seed)
\State \textbf{EvolutionEngine} $\;=\;$ ($X, y$, cfg, rng, counters: birth/ref/eval\_count, eval\_budget, start\_time)
\State \textbf{EngineState} $\;=\;$ (engine, populations, policy\_state, current\_iteration / population / inner\_cycle)
\State tree\_size(n), tree\_height(n): recursive node count / depth.
\State evaluate\_tree(n, $X$): recursive broadcast evaluation with NaN-safe ops (safe\_log, safe\_sqrt, safe\_pow, \dots).
\end{algorithmic}

\section*{Budget}
\begin{algorithmic}[1]
\Function{HasBudget}{engine}
  \State \Return $(\text{eval\_budget} = \varnothing \lor \text{eval\_count} < \text{eval\_budget}) \;\land\; (\text{timeout} = \varnothing \lor \text{elapsed} < \text{timeout})$
\EndFunction
\end{algorithmic}

\section*{Scoring an expression}
\begin{algorithmic}[1]
\Function{MakeIndividual}{tree, state, config, parent\_ref}
  \If{$\lnot$\Call{HasBudget}{engine}} \Return $\varnothing$ \EndIf
  \State eval\_count $\gets$ eval\_count $+ 1$
  \State (loss, cost) $\gets$ \slot{loss\_function}(tree, tree\_size(tree), state, config) \cmt{$\varnothing$ if rejected}
  \State \Return Individual(tree, loss, cost, tree\_size(tree), next\_birth, next\_ref, parent\_ref)
\EndFunction
\end{algorithmic}

\section*{Random tree construction}
\begin{algorithmic}[1]
\Function{RandomTerminal}{}
  \State with prob.\ $\tfrac12$: VarNode(random feature); else ConstNode(random from cfg.constants, or $\mathcal N(0,1)$)
\EndFunction
\Function{AppendRandomOp}{tree}
  \State pick a random leaf; sample arity $\in\{1,2\}$ weighted by \#unary : \#binary ops; sample op of that arity
  \State replace the leaf with OpNode(op, RandomTerminal, [RandomTerminal])
\EndFunction
\State \textit{Also available to policies:} PrependRandomOp (wrap root), InsertRandomOp (wrap random node),
\Statex RandomTreeFixedSize($k$) (append until $k$ nodes), RandomTree(max\_depth, full) (grow/full recursion).
\Function{ValidTree}{tree}
  \State \Return tree\_size $\le$ maxsize $\land$ tree\_height $\le$ maxdepth $\land$ \Call{CheckConstraints}{tree}
\EndFunction
\Function{CheckConstraints}{tree}
  \ForEach{OpNode $n$ in tree}
    \State if constraints[$n$.op] set: child subtree sizes must not exceed the per-argument limits
    \State if nested\_constraints[$n$.op] set: nesting depth of each listed child op under $n$ must not exceed its limit
  \EndFor
  \State \Return true if all pass
\EndFunction
\end{algorithmic}

\section*{Population initialisation}
\begin{algorithmic}[1]
\Function{InitializePopulation}{state, config}
  \State pop $\gets [\,]$
  \While{$|\text{pop}| <$ population\_size $\land$ \Call{HasBudget}{}}
    \State tree $\gets$ ConstNode(0); apply \Call{AppendRandomOp}{} three times
    \If{$\lnot$\Call{ValidTree}{tree}} tree $\gets$ \Call{RandomTree}{$\min(3,\text{maxdepth})$, full=false} \EndIf
    \State $m \gets$ \Call{MakeIndividual}{tree}; push if $m \ne \varnothing$
  \EndWhile
  \If{pop empty} push the constant $\bar y$ (with loss $\infty$ if it cannot be scored) \EndIf
  \State \Return pop
\EndFunction
\end{algorithmic}

\section*{Constant optimisation and simplification}
\begin{algorithmic}[1]
\Function{OptimizeConstants}{member}
  \State consts $\gets$ constant nodes of member.tree; \Return member if none or eval budget $\le 1$
  \State starts $\gets$ \{current values\} $\cup$ \{nrestarts copies perturbed by $\times(1 + 0.5\,\mathcal N(0,1))$\}
  \State algorithm $\gets$ Newton if single constant and enabled, else Nelder--Mead; cap f\_calls by remaining budget
  \ForEach{$x_0 \in$ starts while \Call{HasBudget}{}}
    \State minimise $\;\text{vals} \mapsto \slot{loss\_function}(\text{tree with constants} = \text{vals})\;$ (each call costs one eval)
    \State keep best (tree, loss) if improved; on optimiser error, warn and continue
  \EndFor
  \If{improved} re-score best tree once more; \Return new Individual with parent\_ref = member.ref \EndIf
  \State \Return member
\EndFunction
\Function{SimplifyTree}{tree}
  \State recursively fold OpNode(op, Const, Const) into a Const (clamped to $\pm10^6$) when finite
\EndFunction
\Function{OptimizeAndSimplify!}{population}
  \ForEach{member in population}
    \If{should\_simplify} $s \gets$ \Call{SimplifyTree}{}; adopt if \Call{ValidTree}{$s$} \EndIf
    \If{should\_optimize\_constants $\land$ rand $<$ optimize\_probability} member $\gets$ \Call{OptimizeConstants}{member} \EndIf
  \EndFor
\EndFunction
\end{algorithmic}

\section*{Main loop}
\begin{algorithmic}[1]
\Function{FitSkeletonSR}{$X, y$, config}
  \State state $\gets$ EvolutionEngine($X, y$, cfg) $+$ \slot{init\_state}(config)
  \State state.populations $\gets$ [\Call{InitializePopulation}{} for $1..n_{\text{pops}}$]
  \State \slot{update\_state!}(populations, state, config)
  \For{iteration $= 1 \ldots$ niterations while \Call{HasBudget}{}}
    \ForEach{population index $p$ while \Call{HasBudget}{}}
      \For{inner $= 1 \ldots$ ncycles\_per\_iteration while \Call{HasBudget}{}}
        \State \slot{update\_state!}(populations, state, config)
        \State \Call{EvolveCycle!}{state, $p$}
        \State record trace snapshot if a milestone (in evals or iterations) was crossed
      \EndFor
      \State \Call{OptimizeAndSimplify!}{populations[$p$]} \quad (if budget remains)
      \State \slot{update\_state!}(populations, state, config)
      \State populations $\gets$ \slot{update\_population}(policy\_state, populations, state, config) \cmt{e.g.\ migration, archive}
    \EndFor
  \EndFor
  \State \Return archived (or best) expressions sorted by complexity, plus n\_evals and the trace
\EndFunction
\end{algorithmic}

\begin{algorithmic}[1]
\Function{EvolveCycle!}{state, $p$}
  \State pop $\gets$ populations[$p$]
  \For{$1 \ldots$ \slot{cycles\_per\_population}(pop)}\cmt{default: $\lceil |\text{pop}| / \text{tournament}_n \rceil$ rounds}
    \If{$\lnot$\Call{HasBudget}{}} \textbf{break} \EndIf
    \If{\slot{should\_crossover}(pop)}\cmt{default: rand $\le$ crossover\_probability}
      \State $a \gets$ \slot{selection}(pop); $b \gets$ \slot{selection}(pop); if $a \equiv b$, take the next member as $b$
      \State trees $\gets$ \slot{crossover}($a$, $b$) \cmt{may return 0, 1 or 2 trees}
      \State candidates $\gets$ [child for (tree, parent) in zip(trees, ($a$,$b$)) if child $\gets$ \Call{MakeIndividual}{} and \slot{acceptance}(parent, child)]
      \If{candidates non-empty} pop $\gets$ \slot{survival}(pop, candidates) \EndIf
    \Else
      \State parent $\gets$ \slot{selection}(pop)
      \State result $\gets$ \slot{mutation}(parent)
      \If{result $= \varnothing$} \textbf{continue}
      \ElsIf{result is an Individual} pop $\gets$ \slot{survival}(pop, [result]) \cmt{already scored by the policy}
      \Else
        \State child $\gets$ \Call{MakeIndividual}{result, parent\_ref = parent.ref}; \textbf{return} if $\varnothing$ (budget exhausted)
        \If{\slot{acceptance}(parent, child)} pop $\gets$ \slot{survival}(pop, [child]) \EndIf
      \EndIf
    \EndIf
    \State populations[$p$] $\gets$ pop
  \EndFor
\EndFunction
\end{algorithmic}
